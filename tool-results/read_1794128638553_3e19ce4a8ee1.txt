     1→'use client'
     2→
     3→// THE SONAR ROOM — you are blind. Click to ping, WASD to move.
     4→// The screen is 90% black; the world only exists where sound has touched it.
     5→
     6→import { useCallback, useEffect, useRef, useState, useSyncExternalStore } from 'react'
     7→import { io, type Socket } from 'socket.io-client'
     8→import { SonarEngine, RunResult } from '@/lib/game/engine'
     9→import { SoundEngine } from '@/lib/game/audio'
    10→import type { Role, ReadyPayload, HuntEvt, CreateAck, JoinAck } from '@/lib/game/net'
    11→
    12→const ROOM_SERVICE_PORT = 3003
    13→
    14→type Phase = 'menu' | 'playing' | 'dead' | 'win'
    15→type MpStage = 'off' | 'choose' | 'pick' | 'join' | 'lobby'
    16→
    17→interface Escape {
    18→  timeMs: number
    19→  pings: number
    20→  keysFound: number
    21→  createdAt: string
    22→}
    23→
    24→const BEST_KEY = 'sonar_best_ms'
    25→
    26→function fmt(ms: number): string {
    27→  const s = Math.floor(ms / 1000)
    28→  const m = Math.floor(s / 60)
    29→  return `${String(m).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`
    30→}
    31→
    32→const emptySubscribe = () => () => {}
    33→
    34→export default function SonarRoom() {
    35→  // Browser extensions (Dark Reader, Grammarly, translators…) inject <style>
    36→  // and attribute mutations into the DOM before React hydrates, which breaks
    37→  // structural hydration matching. The game is a pure client-side canvas +
    38→  // WebAudio experience, so the server only needs to paint the black room.
    39→  // Hydrate a minimal, trivially-matchable shell first, then mount the real
    40→  // game once we know we are on the client.
    41→  const hydrated = useSyncExternalStore(
    42→    emptySubscribe,
    43→    () => true,
    44→    () => false,
    45→  )
    46→  if (!hydrated) {
    47→    return <div className="fixed inset-0 bg-black" suppressHydrationWarning />
    48→  }
    49→  return <SonarRoomGame />
    50→}
    51→
    52→function SonarRoomGame() {
    53→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
    54→  const wrapRef = useRef<HTMLDivElement | null>(null)
    55→  const engineRef = useRef<SonarEngine | null>(null)
    56→  const audioRef = useRef<SoundEngine | null>(null)
    57→  const phaseRef = useRef<Phase>('menu')
    58→  const keysStateRef = useRef<Set<string>>(new Set())
    59→  const joyRef = useRef({ x: 0, y: 0, active: false })
    60→  const joyIdRef = useRef<number | null>(null)
    61→  const joyCenterRef = useRef({ x: 0, y: 0 })
    62→  const timerRef = useRef<HTMLSpanElement | null>(null)
    63→  const knobRef = useRef<HTMLDivElement | null>(null)
    64→  const toastTimeoutRef = useRef<number>(0)
    65→  const endDelayRef = useRef<number>(0)
    66→  const netRef = useRef<Socket | null>(null)
    67→  const roleRef = useRef<Role | null>(null)
    68→  const isHostRef = useRef(false)
    69→  const mpStageRef = useRef<MpStage>('off')
    70→  const mpActiveRef = useRef(false)
    71→  const iWantRef = useRef(false)
    72→  const peerWantsRef = useRef(false)
    73→  const peerLinkedRef = useRef(false)
    74→  const startTimerRef = useRef<number>(0)
    75→
    76→  const [phase, setPhaseState] = useState<Phase>('menu')
    77→  const [keysFound, setKeysFound] = useState(0)
    78→  const [toast, setToast] = useState<string | null>(null)
    79→  const [result, setResult] = useState<RunResult | null>(null)
    80→  const [best, setBest] = useState<number | null>(null)
    81→  const [escapes, setEscapes] = useState<Escape[]>([])
    82→  const [touchMode, setTouchMode] = useState(false)
    83→  const [menuVisible, setMenuVisible] = useState(true)
    84→  const [mpStage, setMpStage] = useState<MpStage>('off')
    85→  const [role, setRole] = useState<Role | null>(null)
    86→  const [roomCode, setRoomCode] = useState<string | null>(null)
    87→  const [mpStatus, setMpStatus] = useState('')
    88→  const [mpError, setMpError] = useState('')
    89→  const [joinInput, setJoinInput] = useState('')
    90→  const [peerLinked, setPeerLinked] = useState(false)
    91→  const [peerLost, setPeerLost] = useState(false)
    92→  const [iWant, setIWant] = useState(false)
    93→  const [peerWants, setPeerWants] = useState(false)
    94→  const [copied, setCopied] = useState(false)
    95→
    96→  const setMpStageBoth = useCallback((s: MpStage) => {
    97→    mpStageRef.current = s
    98→    setMpStage(s)
    99→  }, [])
   100→
   101→  const setPhase = useCallback((p: Phase) => {
   102→    phaseRef.current = p
   103→    setPhaseState(p)
   104→  }, [])
   105→
   106→  const showToast = useCallback((msg: string) => {
   107→    setToast(msg)
   108→    window.clearTimeout(toastTimeoutRef.current)
   109→    toastTimeoutRef.current = window.setTimeout(() => setToast(null), 2600)
   110→  }, [])
   111→
   112→  // ------------------------------------------------------------- joystick
   113→  const resetJoy = useCallback(() => {
   114→    joyIdRef.current = null
   115→    joyRef.current = { x: 0, y: 0, active: false }
   116→    if (knobRef.current) knobRef.current.style.transform = 'translate(-50%, -50%)'
   117→  }, [])
   118→
   119→  const loadEscapes = useCallback(async () => {
   120→    try {
   121→      const res = await fetch('/api/runs', { cache: 'no-store' })
   122→      if (!res.ok) return
   123→      const data = (await res.json()) as { escapes?: Escape[] }
   124→      setEscapes(data.escapes ?? [])
   125→    } catch {
   126→      /* offline is fine */
   127→    }
   128→  }, [])
   129→
   130→  const handleEnd = useCallback(
   131→    (r: RunResult, mp = false) => {
   132→      setResult(r)
   133→      // Let the death flash / win tone breathe before the overlay lands.
   134→      window.clearTimeout(endDelayRef.current)
   135→      endDelayRef.current = window.setTimeout(() => {
   136→        setPhase(r.outcome === 'escaped' ? 'win' : 'dead')
   137→        setMenuVisible(true)
   138→      }, r.outcome === 'escaped' ? 700 : 1100)
   139→      if (mp) {
   140→        // Multiplayer runs are not comparable with solo times — share the
   141→        // verdict with the hunter instead of the leaderboard.
   142→        netRef.current?.emit('hunt:evt', {
   143→          t: 'end',
   144→          outcome: r.outcome,
   145→          timeMs: r.timeMs,
   146→          keysFound: r.keysFound,
   147→          pings: r.pings,
   148→        } satisfies HuntEvt)
   149→        return
   150→      }
   151→      // Persist.
   152→      try {
   153→        if (r.outcome === 'escaped') {
   154→          const prev = window.localStorage.getItem(BEST_KEY)
   155→          const prevN = prev ? Number(prev) : Infinity
   156→          if (r.timeMs < prevN) {
   157→            window.localStorage.setItem(BEST_KEY, String(r.timeMs))
   158→            setBest(r.timeMs)
   159→          }
   160→        }
   161→      } catch {
   162→        /* storage may be unavailable */
   163→      }
   164→      void fetch('/api/runs', {
   165→        method: 'POST',
   166→        headers: { 'Content-Type': 'application/json' },
   167→        body: JSON.stringify(r),
   168→      })
   169→        .then(() => (r.outcome === 'escaped' ? loadEscapes() : undefined))
   170→        .catch(() => undefined)
   171→    },
   172→    [loadEscapes, setPhase],
   173→  )
   174→
   175→  const startGame = useCallback((seed?: number) => {
   176→    const audio = audioRef.current
   177→    const engine = engineRef.current
   178→    if (!audio || !engine) return
   179→    // Touch-capable devices get the joystick from the very first frame —
   180→    // never make a blind player tap first just to unlock their controls.
   181→    const coarse = window.matchMedia?.('(pointer: coarse)').matches ?? false
   182→    const touchable = (navigator.maxTouchPoints ?? 0) > 0 || 'ontouchstart' in window
   183→    if (coarse || touchable) setTouchMode(true)
   184→    // Multiplayer: role decides who simulates what. Solo: vs the AI thing.
   185→    const mp = mpActiveRef.current || (mpStageRef.current === 'lobby' && roleRef.current !== null)
   186→    mpActiveRef.current = mp
   187→    engine.mode = mp ? (roleRef.current === 'monster' ? 'hunter' : 'host') : 'solo'
   188→    iWantRef.current = false
   189→    peerWantsRef.current = false
   190→    setIWant(false)
   191→    setPeerWants(false)
   192→    setPeerLost(false)
   193→    setMpStageBoth('off')
   194→    audio.ensure()
   195→    window.clearTimeout(endDelayRef.current)
   196→    resetJoy()
   197→    setResult(null)
   198→    setKeysFound(0)
   199→    setToast(null)
   200→    setMenuVisible(false)
   201→    engine.start(seed ?? Math.floor(Math.random() * 1e9))
   202→    setPhase('playing')
   203→  }, [resetJoy, setPhase, setMpStageBoth])
   204→
   205→  // --------------------------------------------------------- multiplayer
   206→  const exitMp = useCallback(() => {
   207→    netRef.current?.emit('hunt:leave')
   208→    roleRef.current = null
   209→    isHostRef.current = false
   210→    mpActiveRef.current = false
   211→    iWantRef.current = false
   212→    peerWantsRef.current = false
   213→    setRole(null)
   214→    setRoomCode(null)
   215→    setMpStatus('')
   216→    setMpError('')
   217→    setPeerLinked(false)
   218→    setPeerLost(false)
   219→    setIWant(false)
   220→    setPeerWants(false)
   221→    setJoinInput('')
   222→    setMpStageBoth('off')
   223→    setPhase('menu')
   224→    setMenuVisible(true)
   225→  }, [setPhase, setMpStageBoth])
   226→
   227→  const beginCreate = useCallback(
   228→    (r: Role) => {
   229→      setMpError('')
   230→      roleRef.current = r
   231→      setRole(r)
   232→      isHostRef.current = true
   233→      const socket = netRef.current
   234→      if (!socket) return
   235→      socket.emit('hunt:create', { role: r }, (ack: CreateAck) => {
   236→        if (ack?.ok && ack.code) {
   237→          setRoomCode(ack.code)
   238→          setMpStatus('WAITING IN THE DARK — SHARE THE CODE')
   239→          setMpStageBoth('lobby')
   240→        } else {
   241→          setMpError(ack?.error ?? 'THE DEEP IS SILENT — TRY AGAIN')
   242→        }
   243→      })
   244→    },
   245→    [],
   246→  )
   247→
   248→  const doJoin = useCallback(() => {
   249→    setMpError('')
   250→    const socket = netRef.current
   251→    if (!socket) return
   252→    socket.emit('hunt:join', { code: joinInput }, (ack: JoinAck) => {
   253→      if (ack?.ok && ack.code && ack.role) {
   254→        roleRef.current = ack.role
   255→        setRole(ack.role)
   256→        isHostRef.current = false
   257→        setRoomCode(ack.code)
   258→        setMpStatus('FOUND THEM — LINKING…')
   259→        setMpStageBoth('lobby')
   260→      } else {
   261→        setMpError(ack?.error ?? 'NO ANSWER — TRY AGAIN')
   262→      }
   263→    })
   264→  }, [joinInput])
   265→
   266→  const requestRematch = useCallback(() => {
   267→    if (!mpActiveRef.current) {
   268→      startGame()
   269→      return
   270→    }
   271→    iWantRef.current = true
   272→    setIWant(true)
   273→    netRef.current?.emit('hunt:evt', { t: 'rematch' } satisfies HuntEvt)
   274→    if (peerWantsRef.current && isHostRef.current) {
   275→      const seed = Math.floor(Math.random() * 1e9)
   276→      netRef.current?.emit('hunt:evt', { t: 'start', seed } satisfies HuntEvt)
   277→      startGame(seed)
   278→    }
   279→  }, [startGame])
   280→
   281→  // ------------------------------------------------------------ bootstrap
   282→  useEffect(() => {
   283→    const audio = new SoundEngine()
   284→    audioRef.current = audio
   285→    const engine = new SonarEngine(audio, {
   286→      onKeys: (n) => {
   287→        setKeysFound(n)
   288→        if (mpActiveRef.current) {
   289→          netRef.current?.emit('hunt:evt', { t: 'keys', n } satisfies HuntEvt)
   290→        }
   291→      },
   292→      onToast: showToast,
   293→      onEnd: (r) => handleEnd(r, mpActiveRef.current),
   294→      onEmit: (e) => netRef.current?.emit('hunt:evt', e as HuntEvt),
   295→    })
   296→    engineRef.current = engine
   297→
   298→    // Room service — pairs the two devices and relays game events.
   299→    const socket = io(`/?XTransformPort=${ROOM_SERVICE_PORT}`, {
   300→      transports: ['websocket', 'polling'],
   301→      reconnection: true,
   302→      reconnectionAttempts: Infinity,
   303→      reconnectionDelay: 1000,
   304→      timeout: 10000,
   305→    })
   306→    netRef.current = socket
   307→
   308→    socket.on('connect', () => {
   309→      if (mpStageRef.current === 'lobby' && !peerLinkedRef.current) {
   310→        setMpStatus((s) => (s.startsWith('SIGNAL') ? 'CONTACTING THE DEEP…' : s))
   311→      }
   312→    })
   313→    socket.on('hunt:ready', (p: ReadyPayload) => {
   314→      roleRef.current = p.myRole
   315→      setRole(p.myRole)
   316→      setPeerLinked(true)
   317→      peerLinkedRef.current = true
   318→      setMpStatus('LINKED — DESCENDING…')
   319→      showToast(p.myRole === 'monster' ? 'THE PREY HAS DESCENDED' : 'THE HUNTER IS LISTENING')
   320→      if (isHostRef.current) {
   321→        // The creator's client chooses the shared seed and starts the run.
   322→        window.clearTimeout(startTimerRef.current)
   323→        startTimerRef.current = window.setTimeout(() => {
   324→          const seed = Math.floor(Math.random() * 1e9)
   325→          socket.emit('hunt:evt', { t: 'start', seed } satisfies HuntEvt)
   326→          startGame(seed)
   327→        }, 1500)
   328→      }
   329→    })
   330→    socket.on('hunt:evt', (e: HuntEvt) => {
   331→      if (!e || typeof e !== 'object') return
   332→      switch (e.t) {
   333→        case 'start': {
   334→          if (phaseRef.current !== 'playing') startGame(e.seed)
   335→          break
   336→        }
   337→        case 'pos': {
   338→          if (roleRef.current === 'monster') engine.setRemoteSurvivor(e.x, e.y, e.moving)
   339→          else engine.setRemoteMonster(e.x, e.y, e.moving)
   340→          break
   341→        }
   342→        case 'ping': {
   343→          if (roleRef.current === 'monster') engine.netSurvivorPing(e.x, e.y)
   344→          break
   345→        }
   346→        case 'click': {
   347→          if (roleRef.current === 'survivor') engine.netMonsterClick(e.x, e.y)
   348→          break
   349→        }
   350→        case 'keys': {
   351→          setKeysFound(e.n)
   352→          if (e.n >= 3) {
   353→            audio.unlock()
   354→            showToast('THE DOOR IS OPEN — STOP THEM')
   355→          } else {
   356→            showToast(`THE PREY FOUND A KEY — ${e.n}/3`)
   357→          }
   358→          break
   359→        }
   360→        case 'end': {
   361→          // The survivor's client is authoritative — this verdict is final.
   362→          const iAmHunter = roleRef.current === 'monster'
   363→          const won = iAmHunter ? e.outcome === 'caught' : e.outcome === 'escaped'
   364→          audio.setDanger(0)
   365→          audio.setExitHum(0)
   366→          if (e.outcome === 'caught') audio.win()
   367→          else audio.death()
   368→          window.clearTimeout(endDelayRef.current)
   369→          setResult({ outcome: e.outcome, timeMs: e.timeMs, keysFound: e.keysFound, pings: e.pings })
   370→          endDelayRef.current = window.setTimeout(() => {
   371→            setPhase(won ? 'win' : 'dead')
   372→            setMenuVisible(true)
   373→          }, 1100)
   374→          break
   375→        }
   376→        case 'rematch': {
   377→          peerWantsRef.current = true
   378→          setPeerWants(true)
   379→          showToast('THE OTHER SIDE WANTS A REMATCH')
   380→          if (iWantRef.current && isHostRef.current) {
   381→            const seed = Math.floor(Math.random() * 1e9)
   382→            socket.emit('hunt:evt', { t: 'start', seed } satisfies HuntEvt)
   383→            startGame(seed)
   384→          }
   385→          break
   386→        }
   387→        case 'peer-lost': {
   388→          peerLinkedRef.current = false
   389→          setPeerLinked(false)
   390→          if (phaseRef.current === 'playing') {
   391→            setPeerLost(true)
   392→          } else {
   393→            setMpStatus('YOUR PARTNER VANISHED INTO THE DARK')
   394→          }
   395→          break
   396→        }
   397→      }
   398→    })
   399→
   400→    try {
   401→      const b = window.localStorage.getItem(BEST_KEY)
   402→      if (b) queueMicrotask(() => setBest(Number(b)))
   403→    } catch {
   404→      /* ignore */
   405→    }
   406→    queueMicrotask(() => void loadEscapes())
   407→
   408→    const canvas = canvasRef.current
   409→    const ctx = canvas?.getContext('2d')
   410→
   411→    const resize = () => {
   412→      if (!canvas) return
   413→      const dpr = window.devicePixelRatio || 1
   414→      const w = canvas.clientWidth || window.innerWidth
   415→      const h = canvas.clientHeight || window.innerHeight
   416→      canvas.width = Math.round(w * dpr)
   417→      canvas.height = Math.round(h * dpr)
   418→    }
   419→    resize()
   420→    window.addEventListener('resize', resize)
   421→
   422→    let raf = 0
   423→    let last = performance.now()
   424→    let netFrame = 0
   425→    const loop = (ts: number) => {
   426→      const dt = Math.min(0.05, (ts - last) / 1000)
   427→      last = ts
   428→      // Merge keyboard + joystick into the engine input.
   429→      const ks = keysStateRef.current
   430→      let kx = 0
   431→      let ky = 0
   432→      if (ks.has('KeyD') || ks.has('d') || ks.has('ArrowRight')) kx += 1
   433→      if (ks.has('KeyA') || ks.has('a') || ks.has('ArrowLeft')) kx -= 1
   434→      if (ks.has('KeyS') || ks.has('s') || ks.has('ArrowDown')) ky += 1
   435→      if (ks.has('KeyW') || ks.has('w') || ks.has('ArrowUp')) ky -= 1
   436→      let ix = kx + joyRef.current.x
   437→      let iy = ky + joyRef.current.y
   438→      const il = Math.hypot(ix, iy)
   439→      if (il > 1) {
   440→        ix /= il
   441→        iy /= il
   442→      }
   443→      engine.input.x = ix
   444→      engine.input.y = iy
   445→
   446→      if (ctx) {
   447→        if (phaseRef.current !== 'menu') engine.update(dt)
   448→        const dpr = window.devicePixelRatio || 1
   449→        const cw = canvas ? canvas.width / dpr : 0
   450→        const ch = canvas ? canvas.height / dpr : 0
   451→        engine.render(ctx, cw, ch, dpr)
   452→      }
   453→      if (timerRef.current && phaseRef.current === 'playing') {
   454→        timerRef.current.textContent = fmt(engine.elapsedMs)
   455→      }
   456→      // Stream the local body to the partner at ~15 Hz.
   457→      netFrame++
   458→      if (
   459→        mpActiveRef.current &&
   460→        phaseRef.current === 'playing' &&
   461→        netFrame % 4 === 0 &&
   462→        netRef.current?.connected
   463→      ) {
   464→        const p = engine.playerPos
   465→        const moving = Math.hypot(engine.input.x, engine.input.y) > 0.08
   466→        netRef.current.emit('hunt:evt', {
   467→          t: 'pos',
   468→          x: Math.round(p.x * 10) / 10,
   469→          y: Math.round(p.y * 10) / 10,
   470→          moving,
   471→        } satisfies HuntEvt)
   472→      }
   473→      // Drop any held controls when the run ends so the player
   474→      // never keeps sliding into the dark after death.
   475→      if (phaseRef.current !== 'playing') resetJoy()
   476→      raf = requestAnimationFrame(loop)
   477→    }
   478→    raf = requestAnimationFrame(loop)
   479→
   480→    const onKeyDown = (e: KeyboardEvent) => {
   481→      if (
   482→        ['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code) ||
   483→        [' ', 'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(e.key.toLowerCase())
   484→      ) {
   485→        e.preventDefault()
   486→      }
   487→      if (e.repeat) return
   488→      keysStateRef.current.add(e.code)
   489→      keysStateRef.current.add(e.key === ' ' ? 'space' : e.key.toLowerCase())
   490→      const k = e.key.toLowerCase()
   491→      if (e.code === 'Space' || e.key === ' ' || k === 'enter') {
   492→        if (phaseRef.current === 'menu') {
   493→          if (mpStageRef.current === 'off') startGame()
   494→        } else if (phaseRef.current === 'playing') engine.ping()
   495→        else {
   496→          if (mpActiveRef.current) requestRematch()
   497→          else startGame()
   498→        }
   499→      }
   500→      if (k === 'r' && (phaseRef.current === 'dead' || phaseRef.current === 'win')) {
   501→        if (mpActiveRef.current) requestRematch()
   502→        else startGame()
   503→      }
   504→    }
   505→    const onKeyUp = (e: KeyboardEvent) => {
   506→      keysStateRef.current.delete(e.code)
   507→      keysStateRef.current.delete(e.key === ' ' ? 'space' : e.key.toLowerCase())
   508→    }
   509→    const onBlur = () => {
   510→      keysStateRef.current.clear()
   511→      joyRef.current = { x: 0, y: 0, active: false }
   512→    }
   513→
   514→    window.addEventListener('keydown', onKeyDown)
   515→    window.addEventListener('keyup', onKeyUp)
   516→    window.addEventListener('blur', onBlur)
   517→
   518→    return () => {
   519→      cancelAnimationFrame(raf)
   520→      window.removeEventListener('resize', resize)
   521→      window.removeEventListener('keydown', onKeyDown)
   522→      window.removeEventListener('keyup', onKeyUp)
   523→      window.removeEventListener('blur', onBlur)
   524→      window.clearTimeout(toastTimeoutRef.current)
   525→      window.clearTimeout(endDelayRef.current)
   526→      window.clearTimeout(startTimerRef.current)
   527→      socket.disconnect()
   528→      audio.dispose()
   529→    }
   530→  }, [handleEnd, loadEscapes, requestRematch, resetJoy, showToast, startGame])
   531→
   532→  // ------------------------------------------------------------- pointer
   533→  const onCanvasPointerDown = (e: React.PointerEvent) => {
   534→    if (e.pointerType === 'touch') setTouchMode(true)
   535→    if (phaseRef.current === 'playing') {
   536→      e.preventDefault()
   537→      engineRef.current?.ping()
   538→    }
   539→  }
   540→
   541→  // ------------------------------------------------------------- joystick
   542→  const onJoyDown = (e: React.PointerEvent) => {
