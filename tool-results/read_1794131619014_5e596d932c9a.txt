     1→// The Sonar Room — core game engine.
     2→// You are blind: the world only exists for the ~1.5s a ping takes to fade.
     3→// Pings expand as wavefronts; where they touch wall segments the wireframe
     4→// blinks into existence. Sound is spatialized. The thing in the dark hears
     5→// every ping you make.

   111→const PING_MAX_R = 640

   114→const PLAYER_SPEED = 152
   115→const PLAYER_R = 9
   116→const SENSE_R = 150 // monster senses you this close, ping or not
   117→const LOSE_R = 310 // break this far for long enough and it loses you
   118→const HEAR_R = 920 // it hears any ping made within this range
   119→const CATCH_R = 20

   125→const clamp = (v: number, a: number, b: number): number => Math.max(a, Math.min(b, v))

   129→  map: GameMap | null = null
   130→  phase: EnginePhase = 'idle'

   132→  cooldown = 0

   144→  private pings: Ping[] = []
   145→  private revealT: Float32Array = new Float32Array(0)
   146→  private revealS: Float32Array = new Float32Array(0)
   147→  private shake = 0
   148→  private hbT = 0

   150→  private thudGate = 0
   151→  private screechGate = 0

   154→  private chaseAnnounced = false
   155→  private deathAt = 0

   157→  private seed = 1

   159→  constructor(
   160→    private audio: SoundEngine,
   161→    private hooks: EngineHooks = {},

   164→  private panOf(x: number): number {
   165→    return clamp((x - this.player.x) / 300, -1, 1)

   167→
   168→  private volByDist(d: number, range: number): number {
   169→    return clamp(1 - d / range, 0, 1)

   172→  start(seed = Math.floor(Math.random() * 1e9)): void {
   173→    this.seed = seed
   174→    // Hunt mode (2-player) builds the two-door variant of the maze.
   175→    this.map = generateMap(seed, this.mode !== 'solo')
   176→    const n = this.map.edges.length
   177→    this.revealT = new Float32Array(n).fill(-99)
   178→    this.revealS = new Float32Array(n)
   179→    // In hunter mode the LOCAL body is the monster; the "monster" slot
   180→    // holds the remote survivor ghost. Audio panning stays relative to
   181→    // the local body either way.
   182→    this.player = this.mode === 'hunter' ? { ...this.map.monsterStart } : { ...this.map.start }
   183→    this.cam = { ...this.player }
   184→    this.keys = this.map.keys.map((k) => ({
   185→      x: k.x,
   186→      y: k.y,
   187→      color: k.color,
   188→      found: false,
   189→      revealT: -99,
   190→      jT: 1 + Math.random() * 2,
   191→    }))
   192→    this.doorState = this.map.doors.map((spec) => ({
   193→      spec,
   194→      unlocked: false,
   195→      progress: 0,
   196→      revealT: -99,
   197→      channeling: false,
   198→      tickT: 0,
   199→      lastStep: -1,
   200→      beaconT: 0,
   201→    }))
   202→    const m: Monster = {
   203→      x: this.mode === 'hunter' ? this.map.start.x : this.map.monsterStart.x,
   204→      y: this.mode === 'hunter' ? this.map.start.y : this.map.monsterStart.y,
   205→      state: 'dormant',
   206→      path: [],
   207→      pathI: 0,
   208→      target: null,
   209→      repath: 0,
   210→      lost: 0,
   211→      growlT: 2 + Math.random() * 3,
   212→      clickT: 3 + Math.random() * 3,
   213→      stepT: 0,
   214→      pendingAt: 0,
   215→      pendingX: 0,
   216→      pendingY: 0,
   217→      revealT: -99,
   218→      lastPX: 0,
   219→      lastPY: 0,
   220→      netX: 0,
   221→      netY: 0,
   222→      netMoving: false,
   223→    }
   224→    m.netX = m.x
   225→    m.netY = m.y
   226→    this.monster = m
   227→    this.pings = []
   228→    this.time = 0
   229→    this.cooldown = 0
   230→    this.pingsUsed = 0
   231→    this.keysFound = 0
   232→    this.danger = 0
   233→    this.shake = 0
   234→    this.thudGate = 0
   235→    this.screechGate = 0
   236→    this.beaconT = 0
   237→    this.graceAnnounced = false
   238→    this.chaseAnnounced = false
   239→    this.phase = 'playing'
   240→    this.audio.setDanger(0)
   241→    this.audio.setExitHum(0)
   242→    this.hooks.onKeys?.({ total: 0, crimson: 0, jade: 0 })
   243→    this.hooks.onToast?.(
   244→      this.mode === 'hunter'
   245→        ? 'SOMETHING FLESH IS IN THE DARK — LISTEN'
   246→        : this.mode === 'host'
   247→          ? 'FIND THE COLORED KEYS — QUIETLY'
   248→          : 'FIND THE KEYS — QUIETLY',
   249→    )
   250→  }
   251→
   252→  /** Emit a sonar ping — or, as the hunter, an echolocation click. Costs noise. */
   253→  ping(): boolean {
   254→    if (this.phase !== 'playing' || this.cooldown > 0 || !this.map) return false
   255→    this.cooldown = PING_COOLDOWN
   256→    this.pingsUsed++
   257→    if (this.mode === 'hunter') {
   258→      this.pings.push(this.makePing(this.player.x, this.player.y, 1, 430, 440))
   259→      this.audio.monsterClick(0, 0.32)
   260→      this.hooks.onEmit?.({ t: 'click', x: this.player.x, y: this.player.y })
   261→      return true
   262→    }
   263→    this.pings.push(this.makePing(this.player.x, this.player.y, 0, PING_MAX_R, PING_SPEED))
   264→    this.audio.ping()
   265→    if (this.mode === 'host') {
   266→      this.hooks.onEmit?.({ t: 'ping', x: this.player.x, y: this.player.y })
   267→    }
   268→    const m = this.monster
   269→    if (this.mode === 'solo' && m && m.state !== 'chase' && this.time > GRACE) {
   270→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   271→      if (d < HEAR_R) {
   272→        m.pendingAt = this.time + 0.35
   273→        m.pendingX = this.player.x
   274→        m.pendingY = this.player.y
   275→      }
   276→    }
   277→    return true
   278→  }

   280→  private makePing(ox: number, oy: number, owner: 0 | 1, maxR: number, speed: number): Ping {
   281→    const jag: number[] = []
   282→    for (let i = 0; i < 48; i++) jag.push((Math.random() - 0.5) * 2)
   283→    return {
   284→      ox,
   285→      oy,
   286→      r: 2,
   287→      maxR,
   288→      speed,
   289→      owner,
   290→      hits: new Uint8Array(this.map ? this.map.edges.length : 0),
   291→      doorHit: false,
   292→      jag,
   293→      dead: false,
   294→    }
   295→  }

   297→  // ------------------------------------------------------------- collision

   299→  private hitsWall(x: number, y: number): boolean {
   300→    const map = this.map
   301→    if (!map) return false
   302→    const c0 = Math.floor((x - PLAYER_R) / map.cell)
   303→    const c1 = Math.floor((x + PLAYER_R) / map.cell)
   304→    const r0 = Math.floor((y - PLAYER_R) / map.cell)
   305→    const r1 = Math.floor((y + PLAYER_R) / map.cell)
   306→    for (let r = r0; r <= r1; r++) {
   307→      for (let c = c0; c <= c1; c++) {
   308→        if (!isSolid(map, c, r)) continue
   309→        const rx = c * map.cell
   310→        const ry = r * map.cell
   311→        const nx = clamp(x, rx, rx + map.cell)
   312→        const ny = clamp(y, ry, ry + map.cell)
   313→        if (Math.hypot(x - nx, y - ny) < PLAYER_R) return true
   314→      }
   315→    }
   316→    return false
   317→  }

   319→  // ---------------------------------------------------------------- update
   320→
   321→  update(dt: number): void {
   322→    this.time += dt
   323→    this.cooldown = Math.max(0, this.cooldown - dt)
   324→    this.thudGate -= dt
   325→    this.screechGate -= dt
   326→    this.shake = Math.max(0, this.shake - dt * 1.6)
   327→
   328→    this.updatePings(dt)
   329→
   330→    if (this.phase === 'playing') {
   331→      this.updatePlayer(dt)
   332→      if (this.mode === 'hunter') {
   333→        this.updateGhost(dt)
   334→        this.updateDanger(dt)
   335→      } else if (this.mode === 'host') {
   336→        this.updateRemoteMonster(dt)
   337→        this.updateKeys(dt)
   338→        this.updateDoors(dt)
   339→        this.updateDanger(dt)
   340→      } else {
   341→        this.updateMonster(dt)
   342→        this.updateKeys(dt)
   343→        this.updateDoors(dt)
   344→        this.updateDanger(dt)
   345→      }
   346→    } else {
   347→      this.danger = Math.max(0, this.danger - dt * 0.8)
   348→      this.audio.setDanger(this.danger)
   349→    }
   350→
   351→    // Camera follows.
   352→    const k = Math.min(1, dt * 4.5)
   353→    this.cam.x += (this.player.x - this.cam.x) * k
   354→    this.cam.y += (this.player.y - this.cam.y) * k
   355→  }
   356→
   357→  private updatePings(dt: number): void {
   358→    const map = this.map
   359→    if (!map) return
   360→    for (const p of this.pings) {
   361→      const rPrev = p.r
   362→      p.r = Math.min(p.maxR, p.r + p.speed * dt)
   363→      if (p.owner === 0 || p.owner === 1) {
   364→        // Edge crossings.
   365→        let soundBudget = 2
   366→        for (let i = 0; i < map.edges.length; i++) {
   367→          if (p.hits[i]) continue
   368→          const e = map.edges[i]
   369→          const mx = (e.x1 + e.x2) / 2
   370→          const my = (e.y1 + e.y2) / 2
   371→          // Cheap reject by bounding box.
   372→          const bx = Math.min(e.x1, e.x2) - p.r
   373→          if (p.ox < bx || p.ox > Math.max(e.x1, e.x2) + p.r) {
   374→            if (Math.abs(p.ox - mx) > p.r) continue
   375→          }
   376→          const d = segDist(p.ox, p.oy, e.x1, e.y1, e.x2, e.y2)
   377→          if (d <= p.r && d > rPrev - 0.5) {
   378→            p.hits[i] = 1
   379→            const strength = clamp(1.08 - d / 720, 0.28, 1) * (p.owner === 1 ? 0.5 : 1)
   380→            if (strength > this.revealS[i] * expFade(this.time - this.revealT[i], 0.55)) {
   381→              this.revealS[i] = strength
   382→              this.revealT[i] = this.time
   383→            }
   384→            if (p.owner === 0 && this.phase === 'playing' && soundBudget > 0 && this.thudGate <= 0) {
   385→              const vol = strength * clamp(1.2 - d / 620, 0.2, 1) * 0.85
   386→              this.audio.thud(this.panOf(mx), vol)
   387→              soundBudget--
   388→              this.thudGate = 0.045
   389→            }
   390→          }
   391→        }
   392→        // Door reveals.
   393→        for (const ds of this.doorState) {
   394→          const dd = Math.hypot(ds.spec.midX - p.ox, ds.spec.midY - p.oy)
   395→          if (dd <= p.r && dd > rPrev - 0.5) {
   396→            ds.revealT = this.time
   397→            if (p.owner === 0 && !p.doorHit && this.phase === 'playing') {
   398→              p.doorHit = true
   399→              this.audio.doorTone(this.panOf(ds.spec.midX), this.volByDist(dd, 640) * 0.5 + 0.05)
   400→            }
   401→          }
   402→        }
   403→        // Key reveals.
   404→        for (const k of this.keys) {
   405→          if (k.found) continue
   406→          const kd = Math.hypot(k.x - p.ox, k.y - p.oy)
   407→          if (kd <= p.r && kd > rPrev - 0.5) {
   408→            k.revealT = this.time
   409→            if (p.owner === 0 && this.phase === 'playing') {
   410→              this.audio.jingle(this.panOf(k.x), 0.16)
   411→            }
   412→          }
   413→        }
   414→        // Monster / survivor-ghost reveal (only survivor pings screech).
   415→        const m = this.monster
   416→        if (m && p.owner === 0) {
   417→          const md = Math.hypot(m.x - p.ox, m.y - p.oy)
   418→          if (md <= p.r && md > rPrev - 0.5) {
   419→            m.revealT = this.time
   420→            if (this.phase === 'playing' && this.mode !== 'hunter' && this.screechGate <= 0) {
   421→              this.audio.screech(this.panOf(m.x), clamp(0.6 - md / 1400, 0.22, 0.6))
   422→              this.screechGate = 1.5
   423→              if (this.mode === 'solo' && m.state !== 'chase' && this.time > GRACE) {
   424→                m.pendingAt = this.time + 0.3
   425→                m.pendingX = this.player.x
   426→                m.pendingY = this.player.y
   427→              }
   428→            }
   429→          }
   430→        }
   431→      }
   432→      if (p.r >= p.maxR) p.dead = true
   433→    }
   434→    this.pings = this.pings.filter((p) => !p.dead)
   435→  }

   437→  private updatePlayer(dt: number): void {
   438→    let { x, y } = this.input
   439→    const l = Math.hypot(x, y)
   440→    if (l > 1) {
   441→      x /= l
   442→      y /= l
   443→    }
   444→    const moving = l > 0.08
   445→    const sp = PLAYER_SPEED
   446→    const nx = this.player.x + x * sp * dt
   447→    if (!this.hitsWall(nx, this.player.y)) this.player.x = nx
   448→    const ny = this.player.y + y * sp * dt
   449→    if (!this.hitsWall(this.player.x, ny)) this.player.y = ny
   450→
   451→    if (moving) {
   452→      this.stepT -= dt
   453→      if (this.stepT <= 0) {
   454→        this.stepT = this.mode === 'hunter' ? 0.3 : 0.34
   455→        if (this.mode === 'hunter') this.audio.monsterStep(0, 0.05)
   456→        else this.audio.step(0.045)
   457→      }
   458→    }
   459→  }
   460→
   461→  private updateKeys(dt: number): void {
   462→    for (const k of this.keys) {
   463→      if (k.found) continue
   464→      const d = Math.hypot(this.player.x - k.x, this.player.y - k.y)
   465→      // Faint jingle hints when a key is close by.
   466→      k.jT -= dt
   467→      if (d < 300 && k.jT <= 0) {
   468→        k.jT = 3.1 + Math.random() * 1.6
   469→        this.audio.jingle(this.panOf(k.x), 0.05 + 0.1 * this.volByDist(d, 300))
   470→      }
   471→      if (d < 24) {
   472→        k.found = true
   473→        this.keysFound++
   474→        this.audio.chime(this.panOf(k.x), 0.5)
   475→        const counts = this.counts()
   476→        this.hooks.onKeys?.(counts)
   477→        if (this.mode === 'solo') {
   478→          if (this.keysFound >= 3) {
   479→            const d0 = this.doorState[0]
   480→            if (d0) d0.unlocked = true
   481→            this.audio.unlock()
   482→            this.hooks.onToast?.('THE DOOR IS OPEN')
   483→            const m = this.monster
   484→            if (m && m.state === 'dormant') m.state = 'wander'
   485→          }
   486→        } else if (k.color !== 'white' && counts[k.color] >= 3) {
   487→          this.hooks.onToast?.(`ALL ${k.color.toUpperCase()} KEYS — THAT DOOR CAN BE UNLOCKED`)
   488→        }
   489→      }
   490→    }
   491→  }
   492→
   493→  private counts(): KeyCounts {
   494→    let crimson = 0
   495→    let jade = 0
   496→    for (const k of this.keys) {
   497→      if (!k.found) continue
   498→      if (k.color === 'crimson') crimson++
   499→      else if (k.color === 'jade') jade++
   500→    }
   501→    return { total: this.keysFound, crimson, jade }
   502→  }
   503→
   504→  private keysOf(color: DoorColor): number {
   505→    let n = 0
   506→    for (const k of this.keys) if (k.found && k.color === color) n++
   507→    return n
   508→  }

   510→  private updateDoors(dt: number): void {
   511→    const map = this.map
   512→    if (!map) return
   513→    let hum = 0
   514→    for (let i = 0; i < this.doorState.length; i++) {
   515→      const ds = this.doorState[i]
   516→      const d = Math.hypot(this.player.x - ds.spec.midX, this.player.y - ds.spec.midY)
   517→
   518→      if (ds.unlocked) {
   519→        // Open door: hum + beacon pings, and the way out.
   520→        hum = Math.max(hum, this.volByDist(d, 300))
   521→        ds.beaconT -= dt
   522→        if (ds.beaconT <= 0) {
   523→          ds.beaconT = 2.4
   524→          this.pings.push(this.makePing(ds.spec.midX, ds.spec.midY, 1, 170, 320))
   525→          if (d < 620)
   526→            this.audio.doorBlip(this.panOf(ds.spec.midX), 0.05 + 0.14 * this.volByDist(d, 620))
   527→        }
   528→        if (this.phase === 'playing' && d < (this.mode === 'solo' ? 42 : ESCAPE_R)) {
   529→          this.finishEscape(i)
   530→        }
   531→        continue
   532→      }
   533→
   534→      if (this.mode !== 'host') continue
   535→
   536→      // --- Hunt lock channel: all 3 keys of this door's color, then hold
   537→      //     still beside it for three full seconds. Step away and the lock
   538→      //     slams back shut. ---
   539→      const ready = this.keysOf(ds.spec.color) >= 3
   540→      if (ready && d < UNLOCK_R) {
   541→        if (!ds.channeling) {
   542→          ds.channeling = true
   543→          ds.tickT = 0
   544→          this.hooks.onToast?.('UNLOCKING — HOLD STILL')
   545→        }
   546→        ds.progress += dt / UNLOCK_TIME
   547→        ds.tickT -= dt
   548→        if (ds.tickT <= 0) {
   549→          ds.tickT = 0.75
   550→          this.audio.doorBlip(0, 0.1 + 0.22 * ds.progress)
   551→        }
   552→        const step = Math.floor(ds.progress * 8)
   553→        if (step !== ds.lastStep) {
   554→          ds.lastStep = step
   555→          this.emitDoor(i, ds)
   556→        }
   557→        if (ds.progress >= 1) {
   558→          ds.progress = 1
   559→          ds.unlocked = true
   560→          ds.channeling = false
   561→          this.audio.unlock(this.panOf(ds.spec.midX))
   562→          this.hooks.onToast?.(`THE ${ds.spec.color.toUpperCase()} DOOR IS OPEN — GO`)
   563→          this.emitDoor(i, ds)
   564→        }
   565→      } else {
   566→        if (ds.channeling && ds.progress > 0.08) {
   567→          this.hooks.onToast?.('UNLOCK INTERRUPTED')
   568→          this.audio.thud(this.panOf(ds.spec.midX), 0.3)
   569→        }
   570→        if (ds.channeling && ds.progress > 0) {
   571→          ds.progress = 0
   572→          ds.lastStep = -1
   573→          this.emitDoor(i, ds)
   574→        }
   575→        ds.channeling = false
   576→      }
   577→
   578→      // A lock you can work calls to you — a faint beacon marks the door.
   579→      if (ready) {
   580→        ds.beaconT -= dt
   581→        if (ds.beaconT <= 0) {
   582→          ds.beaconT = 3.4
   583→          this.pings.push(this.makePing(ds.spec.midX, ds.spec.midY, 1, 190, 300))
   584→          if (d < 520)
   585→            this.audio.doorBlip(this.panOf(ds.spec.midX), 0.03 + 0.09 * this.volByDist(d, 520))
   586→        }
   587→      }
   588→    }
   589→    this.audio.setExitHum(hum)
   590→  }

   594→    this.winAt = this.time

   602→      keysFound: this.keysFound,
   603→      pings: this.pingsUsed,
   604→      door: this.doorState[doorIndex]?.spec.color,
   605→    })

   620→  private updateDanger(dt: number): void {
   621→    const m = this.monster
   622→    if (!m) return
   623→    if (this.time > GRACE) {
   624→      if (!this.graceAnnounced) {
   625→        this.graceAnnounced = true
   626→        this.hooks.onToast?.(
   627→          this.mode === 'solo'
   628→            ? 'SOMETHING IS AWAKE'
   629→            : this.mode === 'hunter'
   630→              ? 'THE PREY IS SOMEWHERE — IT PINGS'
   631→              : 'IT IS SOMEWHERE — LISTENING',
   632→        )
   633→        this.audio.growl(this.panOf(m.x), 0.5)
   634→      }
   635→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   636→      let target = clamp(1 - d / 340, 0, 1)
   637→      if (m.state !== 'chase') target *= 0.6
   638→      this.danger += (target - this.danger) * Math.min(1, dt * 4)
   639→      if (this.danger > 0.04) {
   640→        this.hbT -= dt
   641→        if (this.hbT <= 0) {
   642→          this.hbT = 1.15 - 0.68 * this.danger
   643→          this.audio.heartbeat(0.1 + 0.45 * this.danger)
   644→        }
   645→      }
   646→      if (m.state === 'chase') this.shake = Math.max(this.shake, this.danger * 0.55)
   647→    }
   648→    this.audio.setDanger(this.danger)
   649→  }

   653→  private updateMonster(dt: number): void {
   654→    const map = this.map
   655→    const m = this.monster
   656→    if (!map || !m) return
   657→    if (m.state === 'dormant') {
   658→      if (this.time > GRACE) m.state = 'wander'
   659→      return
   660→    }
   661→
   662→    const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   663→    const mc = worldToCell(map, m.x, m.y)
   664→
   665→    // Sense the player up close — no ping needed.
   666→    if (d < SENSE_R && m.state !== 'chase') {
   667→      m.state = 'chase'
   668→      m.path = []
   669→      m.repath = 0
   670→      m.lost = 0
   671→      if (this.screechGate <= 0) {
   672→        this.audio.screech(this.panOf(m.x), 0.65)
   673→        this.screechGate = 1.2
   674→      }
   675→      if (!this.chaseAnnounced) {
   676→        this.chaseAnnounced = true
   677→        this.hooks.onToast?.('RUN')
   678→      }
   679→    }
   680→
   681→    // A ping it heard (delayed reaction).
   682→    if (m.pendingAt > 0 && this.time >= m.pendingAt) {
   683→      m.pendingAt = 0
   684→      if (m.state !== 'chase') {
   685→        const cell = nearestOpenCell(map, ...this.cellOfPoint(m.pendingX, m.pendingY))
   686→        m.state = 'investigate'
   687→        m.target = cell
   688→        m.path = bfsPath(map, mc.c, mc.r, cell.c, cell.r)
   689→        m.pathI = 0
   690→      }
   691→    }
   692→
   693→    let speed = 60
   694→    if (m.state === 'chase') {
   695→      speed = 148 + this.keysFound * 8
   696→      m.lastPX = this.player.x
   697→      m.lastPY = this.player.y
   698→      if (d > LOSE_R) {
   699→        m.lost += dt
   700→        if (m.lost > 2.2) {
   701→          m.state = 'investigate'
   702→          m.target = nearestOpenCell(map, ...this.cellOfPoint(m.lastPX, m.lastPY))
   703→          m.path = bfsPath(map, mc.c, mc.r, m.target.c, m.target.r)
   704→          m.pathI = 0
   705→        }
   706→      } else {
   707→        m.lost = 0
   708→      }
   709→      m.repath -= dt
   710→      const pc = worldToCell(map, this.player.x, this.player.y)
   711→      if (m.repath <= 0) {
   712→        m.repath = 0.32
   713→        m.path = bfsPath(map, mc.c, mc.r, pc.c, pc.r)
   714→        m.pathI = 0
   715→      }
   716→      m.clickT -= dt
   717→      if (m.clickT <= 0) {
   718→        m.clickT = 2.6
   719→        this.monsterClick(m)
   720→      }
   721→      m.stepT -= dt
   722→      if (m.stepT <= 0 && d < 460) {
   723→        m.stepT = 0.27
   724→        this.audio.monsterStep(this.panOf(m.x), 0.1 + 0.3 * this.volByDist(d, 460))
   725→      }
   726→    } else if (m.state === 'investigate') {
   727→      speed = 98
   728→      if (!m.target || m.pathI >= m.path.length) {
   729→        m.state = 'wander'
   730→        m.target = null
   731→        m.path = []
   732→      }
   733→      m.clickT -= dt
   734→      if (m.clickT <= 0) {
   735→        m.clickT = 4.5 + Math.random() * 2.5
   736→        this.monsterClick(m)
   737→      }
   738→    } else {
   739→      // wander — it drifts through the dark, but its instinct pulls it
   740→      // toward your region of the maze (biased random walk).
   741→      speed = 60
   742→      if (m.pathI >= m.path.length) {
   743→        const cands: Cell[] = []
   744→        for (let i = 0; i < 8; i++) {
   745→          const c = clamp(mc.c + Math.floor((Math.random() - 0.5) * 22), 1, map.cols - 2)
   746→          const r = clamp(mc.r + Math.floor((Math.random() - 0.5) * 16), 1, map.rows - 2)
   747→          const cell = nearestOpenCell(map, c, r)
   748→          if (Math.hypot(cell.c - mc.c, cell.r - mc.r) >= 3) cands.push(cell)
   749→        }
   750→        if (cands.length > 0) {
   751→          let pick = cands[Math.floor(Math.random() * cands.length)]
   752→          if (Math.random() < 0.6) {
   753→            const pc = worldToCell(map, this.player.x, this.player.y)
   754→            for (const c of cands) {
   755→              if (
   756→                Math.hypot(c.c - pc.c, c.r - pc.r) <
   757→                Math.hypot(pick.c - pc.c, pick.r - pc.r)
   758→              ) {
   759→                pick = c
   760→              }
   761→            }
   762→          }
   763→          m.path = bfsPath(map, mc.c, mc.r, pick.c, pick.r)
   764→          m.pathI = 0
   765→        }
   766→      }
   767→      m.growlT -= dt
   768→      if (m.growlT <= 0) {
   769→        m.growlT = 3.5 + Math.random() * 4.5
   770→        this.audio.growl(this.panOf(m.x), 0.06 + 0.4 * this.volByDist(d, 560))
   771→      }
   772→      m.clickT -= dt
   773→      if (m.clickT <= 0) {
   774→        m.clickT = 6 + Math.random() * 3
   775→        this.monsterClick(m)
   776→      }
   777→    }
   778→
   779→    // Follow path.
   780→    if (m.pathI < m.path.length) {
   781→      const wp = m.path[m.pathI]
   782→      const dx = wp.x - m.x
   783→      const dy = wp.y - m.y
   784→      const dl = Math.hypot(dx, dy)
   785→      if (dl < 6) {
   786→        m.pathI++
   787→      } else {
   788→        m.x += (dx / dl) * speed * dt
   789→        m.y += (dy / dl) * speed * dt
   790→      }
   791→    }
   792→
   793→    // Catch.
   794→    this.checkCatch(d)
   795→  }

   797→  private checkCatch(d: number): void {
   798→    if (d >= CATCH_R || this.phase !== 'playing') return
   799→    const m = this.monster
   800→    this.deathAt = this.time
   801→    this.phase = 'dead'
   802→    this.shake = 1.4
   803→    this.audio.setDanger(0)
   804→    this.audio.setExitHum(0)
   805→    this.audio.death()
   806→    if (m) m.revealT = this.time
   807→    this.hooks.onEnd?.({
   808→      outcome: 'caught',
   809→      timeMs: Math.round(this.time * 1000),
   810→      keysFound: this.keysFound,
   811→      pings: this.pingsUsed,
   812→    })
   813→  }

   924→  render(ctx: CanvasRenderingContext2D, cw: number, ch: number, dpr: number): void {
   925→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   926→    ctx.fillStyle = '#000000'
   927→    ctx.fillRect(0, 0, cw, ch)
   928→    const map = this.map
   929→    if (!map) return
   930→    const t = this.time
   931→
   932→    const zoom = clamp(Math.min(cw, ch) / 660, 0.9, 1.5)
   933→    const sh = this.shake
   934→    const sx = (Math.random() - 0.5) * 12 * sh
   935→    const sy = (Math.random() - 0.5) * 12 * sh
   936→
   937→    ctx.save()
   938→    ctx.translate(cw / 2 + sx, ch / 2 + sy)
   939→    ctx.scale(zoom, zoom)
   940→    ctx.translate(-this.cam.x, -this.cam.y)
   941→    ctx.lineCap = 'round'
   942→    ctx.lineJoin = 'round'
   943→
   944→    // --- Revealed wall wireframes ---
   945→    for (let i = 0; i < map.edges.length; i++) {
   946→      const age = t - this.revealT[i]
   947→      if (age > 3 || age < 0) continue
   948→      const a = this.revealS[i] * expFade(age, 0.55)
   949→      if (a < 0.015) continue
   950→      const e = map.edges[i]
   951→      ctx.strokeStyle = `rgba(255,255,255,${(a * 0.22).toFixed(3)})`
   952→      ctx.lineWidth = 5
   953→      ctx.beginPath()
   954→      ctx.moveTo(e.x1, e.y1)
   955→      ctx.lineTo(e.x2, e.y2)
   956→      ctx.stroke()
   957→      ctx.strokeStyle = `rgba(255,255,255,${a.toFixed(3)})`
   958→      ctx.lineWidth = 1.8
   959→      ctx.beginPath()
   960→      ctx.moveTo(e.x1, e.y1)
   961→      ctx.lineTo(e.x2, e.y2)
   962→      ctx.stroke()
   963→    }
   964→
   965→    // --- Exit doors (solo: one white · hunt: crimson + jade) ---
   966→    const doorRgb = (c: DoorColor): [number, number, number] =>
   967→      c === 'crimson' ? [255, 70, 86] : c === 'jade' ? [86, 255, 168] : [255, 255, 255]
   968→
   969→    for (const ds of this.doorState) {
   970→      const dAge = t - ds.revealT
   971→      const dA = dAge >= 0 ? expFade(dAge, 0.6) : 0
   972→      // A lock being worked burns through the dark on both sides of the wire.
   973→      const chanA = ds.progress > 0 && !ds.unlocked ? 0.42 : 0
   974→      const baseA = Math.max(dA, chanA)
   975→      if (baseA <= 0.02) continue
   976→      const [cr, cg, cb] = doorRgb(ds.spec.color)
   977→      const door = ds.spec
   978→      const depth = 16
   979→      const half = map.cell / 2 - 6
   980→      ctx.save()
   981→      ctx.translate(door.midX, door.midY)
   982→      if (door.vertical) {
   983→        const dir = door.midX < map.w / 2 ? 1 : -1
   984→        const pulse = ds.unlocked ? 0.65 + 0.35 * Math.sin(t * 5) : 1
   985→        const a = baseA * pulse
   986→        ctx.strokeStyle = `rgba(${cr},${cg},${cb},${(a * 0.9).toFixed(3)})`
   987→        ctx.lineWidth = 2
   988→        ctx.strokeRect(dir > 0 ? -depth : 0, -half, depth, half * 2)
   989→        if (!ds.unlocked) {
   990→          // Sealed: cross bars.
   991→          ctx.beginPath()
   992→          ctx.moveTo(dir > 0 ? -depth : 0, -half)
   993→          ctx.lineTo(dir > 0 ? 0 : depth, half)
   994→          ctx.moveTo(dir > 0 ? -depth : 0, half)
   995→          ctx.lineTo(dir > 0 ? 0 : depth, -half)
   996→          ctx.stroke()
   997→        } else {
   998→          ctx.strokeStyle = `rgba(${cr},${cg},${cb},${(a * 0.5).toFixed(3)})`
   999→          ctx.strokeRect(dir > 0 ? -depth + 4 : 4, -half + 5, depth - 8, half * 2 - 10)
  1000→        }
  1001→      } else {
  1002→        const dir = door.midY < map.h / 2 ? 1 : -1
  1003→        const pulse = ds.unlocked ? 0.65 + 0.35 * Math.sin(t * 5) : 1
  1004→        const a = baseA * pulse
  1005→        ctx.strokeStyle = `rgba(${cr},${cg},${cb},${(a * 0.9).toFixed(3)})`
  1006→        ctx.lineWidth = 2
  1007→        ctx.strokeRect(-half, dir > 0 ? -depth : 0, half * 2, depth)
  1008→        if (!ds.unlocked) {
  1009→          ctx.beginPath()
  1010→          ctx.moveTo(-half, dir > 0 ? -depth : 0)
  1011→          ctx.lineTo(half, dir > 0 ? 0 : depth)
  1012→          ctx.moveTo(-half, dir > 0 ? 0 : depth)
  1013→          ctx.lineTo(half, dir > 0 ? -depth : 0)
  1014→          ctx.stroke()
  1015→        } else {
  1016→          ctx.strokeStyle = `rgba(${cr},${cg},${cb},${(a * 0.5).toFixed(3)})`
  1017→          ctx.strokeRect(-half + 5, dir > 0 ? -depth + 4 : 4, half * 2 - 10, depth - 8)
  1018→        }
  1019→      }
  1020→      // Lock-picking progress ring.
  1021→      if (ds.progress > 0 && !ds.unlocked) {
  1022→        ctx.strokeStyle = `rgba(${cr},${cg},${cb},0.95)`
  1023→        ctx.lineWidth = 2.5
  1024→        ctx.beginPath()
  1025→        ctx.arc(0, 0, 30, -Math.PI / 2, -Math.PI / 2 + ds.progress * Math.PI * 2)
  1026→        ctx.stroke()
  1027→      }
  1028→      ctx.restore()
  1029→    }
  1030→
  1031→    // --- Keys (colored to match their door) ---
  1032→    for (const k of this.keys) {
  1033→      if (k.found) continue
  1034→      const age = t - k.revealT
  1035→      if (age < 0 || age > 2.4) continue
  1036→      const a = expFade(age, 0.6)
  1037→      if (a < 0.02) continue
  1038→      const [kr, kg, kb] = doorRgb(k.color)
