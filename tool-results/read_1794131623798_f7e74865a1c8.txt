   164→   164→  private panOf(x: number): number {

   171→   171→
   172→   172→  start(seed = Math.floor(Math.random() * 1e9)): void {
   173→   173→    this.seed = seed
   174→   174→    // Hunt mode (2-player) builds the two-door variant of the maze.
   175→   175→    this.map = generateMap(seed, this.mode !== 'solo')
   176→   176→    const n = this.map.edges.length
   177→   177→    this.revealT = new Float32Array(n).fill(-99)
   178→   178→    this.revealS = new Float32Array(n)
   179→   179→    // In hunter mode the LOCAL body is the monster; the "monster" slot
   180→   180→    // holds the remote survivor ghost. Audio panning stays relative to
   181→   181→    // the local body either way.
   182→   182→    this.player = this.mode === 'hunter' ? { ...this.map.monsterStart } : { ...this.map.start }
   183→   183→    this.cam = { ...this.player }
   184→   184→    this.keys = this.map.keys.map((k) => ({
   185→   185→      x: k.x,
   186→   186→      y: k.y,
   187→   187→      color: k.color,
   188→   188→      found: false,
   189→   189→      revealT: -99,
   190→   190→      jT: 1 + Math.random() * 2,
   191→   191→    }))
   192→   192→    this.doorState = this.map.doors.map((spec) => ({
   193→   193→      spec,
   194→   194→      unlocked: false,
   195→   195→      progress: 0,
   196→   196→      revealT: -99,
   197→   197→      channeling: false,
   198→   198→      tickT: 0,
   199→   199→      lastStep: -1,
   200→   200→      beaconT: 0,
   201→   201→    }))
   202→   202→    const m: Monster = {
   203→   203→      x: this.mode === 'hunter' ? this.map.start.x : this.map.monsterStart.x,
   204→   204→      y: this.mode === 'hunter' ? this.map.start.y : this.map.monsterStart.y,
   205→   205→      state: 'dormant',
   206→   206→      path: [],
   207→   207→      pathI: 0,
   208→   208→      target: null,
   209→   209→      repath: 0,
   210→   210→      lost: 0,
   211→   211→      growlT: 2 + Math.random() * 3,
   212→   212→      clickT: 3 + Math.random() * 3,
   213→   213→      stepT: 0,
   214→   214→      pendingAt: 0,
   215→   215→      pendingX: 0,
   216→   216→      pendingY: 0,
   217→   217→      revealT: -99,
   218→   218→      lastPX: 0,
   219→   219→      lastPY: 0,
   220→   220→      netX: 0,
   221→   221→      netY: 0,
   222→   222→      netMoving: false,
   223→   223→    }
   224→   224→    m.netX = m.x
   225→   225→    m.netY = m.y
   226→   226→    this.monster = m
   227→   227→    this.pings = []
   228→   228→    this.time = 0
   229→   229→    this.cooldown = 0
   230→   230→    this.pingsUsed = 0
   231→   231→    this.keysFound = 0
   232→   232→    this.danger = 0
   233→   233→    this.shake = 0
   234→   234→    this.thudGate = 0
   235→   235→    this.screechGate = 0
   236→   236→    this.beaconT = 0
   237→   237→    this.graceAnnounced = false
   238→   238→    this.chaseAnnounced = false
   239→   239→    this.phase = 'playing'
   240→   240→    this.audio.setDanger(0)
   241→   241→    this.audio.setExitHum(0)
   242→   242→    this.hooks.onKeys?.({ total: 0, crimson: 0, jade: 0 })
   243→   243→    this.hooks.onToast?.(
   244→   244→      this.mode === 'hunter'
   245→   245→        ? 'SOMETHING FLESH IS IN THE DARK — LISTEN'
   246→   246→        : this.mode === 'host'
   247→   247→          ? 'FIND THE COLORED KEYS — QUIETLY'
   248→   248→          : 'FIND THE KEYS — QUIETLY',
   249→   249→    )
   250→   250→  }
   251→   251→
   252→   252→  /** Emit a sonar ping — or, as the hunter, an echolocation click. Costs noise. */
   253→   253→  ping(): boolean {
   254→   254→    if (this.phase !== 'playing' || this.cooldown > 0 || !this.map) return false
   255→   255→    this.cooldown = PING_COOLDOWN
   256→   256→    this.pingsUsed++
   257→   257→    if (this.mode === 'hunter') {
   258→   258→      this.pings.push(this.makePing(this.player.x, this.player.y, 1, 430, 440))
   259→   259→      this.audio.monsterClick(0, 0.32)
   260→   260→      this.hooks.onEmit?.({ t: 'click', x: this.player.x, y: this.player.y })
   261→   261→      return true
   262→   262→    }
   263→   263→    this.pings.push(this.makePing(this.player.x, this.player.y, 0, PING_MAX_R, PING_SPEED))
   264→   264→    this.audio.ping()
   265→   265→    if (this.mode === 'host') {
   266→   266→      this.hooks.onEmit?.({ t: 'ping', x: this.player.x, y: this.player.y })
   267→   267→    }
   268→   268→    const m = this.monster
   269→   269→    if (this.mode === 'solo' && m && m.state !== 'chase' && this.time > GRACE) {
   270→   270→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   271→   271→      if (d < HEAR_R) {
   272→   272→        m.pendingAt = this.time + 0.35
   273→   273→        m.pendingX = this.player.x
   274→   274→        m.pendingY = this.player.y
   275→   275→      }
   276→   276→    }
   277→   277→    return true
   278→   278→  }
   279→   279→
   280→   280→  private makePing(ox: number, oy: number, owner: 0 | 1, maxR: number, speed: number): Ping {
   281→   281→    const jag: number[] = []
   282→   282→    for (let i = 0; i < 48; i++) jag.push((Math.random() - 0.5) * 2)
   283→   283→    return {
   284→   284→      ox,
   285→   285→      oy,
   286→   286→      r: 2,
   287→   287→      maxR,
   288→   288→      speed,
   289→   289→      owner,
   290→   290→      hits: new Uint8Array(this.map ? this.map.edges.length : 0),
   291→   291→      doorHit: false,
   292→   292→      jag,
   293→   293→      dead: false,
   294→   294→    }
   295→   295→  }

   297→   297→  // ------------------------------------------------------------- collision
   298→   298→
   299→   299→  private hitsWall(x: number, y: number): boolean {
   300→   300→    const map = this.map
   301→   301→    if (!map) return false
   302→   302→    const c0 = Math.floor((x - PLAYER_R) / map.cell)
   303→   303→    const c1 = Math.floor((x + PLAYER_R) / map.cell)
   304→   304→    const r0 = Math.floor((y - PLAYER_R) / map.cell)
   305→   305→    const r1 = Math.floor((y + PLAYER_R) / map.cell)
   306→   306→    for (let r = r0; r <= r1; r++) {
   307→   307→      for (let c = c0; c <= c1; c++) {
   308→   308→        if (!isSolid(map, c, r)) continue
   309→   309→        const rx = c * map.cell
   310→   310→        const ry = r * map.cell
   311→   311→        const nx = clamp(x, rx, rx + map.cell)
   312→   312→        const ny = clamp(y, ry, ry + map.cell)
   313→   313→        if (Math.hypot(x - nx, y - ny) < PLAYER_R) return true
   314→   314→      }
   315→   315→    }
   316→   316→    return false
   317→   317→  }

   321→   321→  update(dt: number): void {
   322→   322→    this.time += dt
   323→   323→    this.cooldown = Math.max(0, this.cooldown - dt)
   324→   324→    this.thudGate -= dt
   325→   325→    this.screechGate -= dt
   326→   326→    this.shake = Math.max(0, this.shake - dt * 1.6)
   327→   327→
   328→   328→    this.updatePings(dt)
   329→   329→
   330→   330→    if (this.phase === 'playing') {
   331→   331→      this.updatePlayer(dt)
   332→   332→      if (this.mode === 'hunter') {
   333→   333→        this.updateGhost(dt)
   334→   334→        this.updateDanger(dt)
   335→   335→      } else if (this.mode === 'host') {
   336→   336→        this.updateRemoteMonster(dt)
   337→   337→        this.updateKeys(dt)
   338→   338→        this.updateDoors(dt)
   339→   339→        this.updateDanger(dt)
   340→   340→      } else {
   341→   341→        this.updateMonster(dt)
   342→   342→        this.updateKeys(dt)
   343→   343→        this.updateDoors(dt)
   344→   344→        this.updateDanger(dt)
   345→   345→      }
   346→   346→    } else {
   347→   347→      this.danger = Math.max(0, this.danger - dt * 0.8)
   348→   348→      this.audio.setDanger(this.danger)
   349→   349→    }
   350→   350→
   351→   351→    // Camera follows.
   352→   352→    const k = Math.min(1, dt * 4.5)
   353→   353→    this.cam.x += (this.player.x - this.cam.x) * k
   354→   354→    this.cam.y += (this.player.y - this.cam.y) * k
   355→   355→  }
   356→   356→
   357→   357→  private updatePings(dt: number): void {
   358→   358→    const map = this.map
   359→   359→    if (!map) return
   360→   360→    for (const p of this.pings) {
   361→   361→      const rPrev = p.r
   362→   362→      p.r = Math.min(p.maxR, p.r + p.speed * dt)
   363→   363→      if (p.owner === 0 || p.owner === 1) {
   364→   364→        // Edge crossings.
   365→   365→        let soundBudget = 2
   366→   366→        for (let i = 0; i < map.edges.length; i++) {
   367→   367→          if (p.hits[i]) continue
   368→   368→          const e = map.edges[i]
   369→   369→          const mx = (e.x1 + e.x2) / 2
   370→   370→          const my = (e.y1 + e.y2) / 2
   371→   371→          // Cheap reject by bounding box.
   372→   372→          const bx = Math.min(e.x1, e.x2) - p.r
   373→   373→          if (p.ox < bx || p.ox > Math.max(e.x1, e.x2) + p.r) {
   374→   374→            if (Math.abs(p.ox - mx) > p.r) continue
   375→   375→          }
   376→   376→          const d = segDist(p.ox, p.oy, e.x1, e.y1, e.x2, e.y2)
   377→   377→          if (d <= p.r && d > rPrev - 0.5) {
   378→   378→            p.hits[i] = 1
   379→   379→            const strength = clamp(1.08 - d / 720, 0.28, 1) * (p.owner === 1 ? 0.5 : 1)
   380→   380→            if (strength > this.revealS[i] * expFade(this.time - this.revealT[i], 0.55)) {
   381→   381→              this.revealS[i] = strength
   382→   382→              this.revealT[i] = this.time
   383→   383→            }
   384→   384→            if (p.owner === 0 && this.phase === 'playing' && soundBudget > 0 && this.thudGate <= 0) {
   385→   385→              const vol = strength * clamp(1.2 - d / 620, 0.2, 1) * 0.85
   386→   386→              this.audio.thud(this.panOf(mx), vol)
   387→   387→              soundBudget--
   388→   388→              this.thudGate = 0.045
   389→   389→            }
   390→   390→          }
   391→   391→        }
   392→   392→        // Door reveals.
   393→   393→        for (const ds of this.doorState) {
   394→   394→          const dd = Math.hypot(ds.spec.midX - p.ox, ds.spec.midY - p.oy)
   395→   395→          if (dd <= p.r && dd > rPrev - 0.5) {
   396→   396→            ds.revealT = this.time
   397→   397→            if (p.owner === 0 && !p.doorHit && this.phase === 'playing') {
   398→   398→              p.doorHit = true
   399→   399→              this.audio.doorTone(this.panOf(ds.spec.midX), this.volByDist(dd, 640) * 0.5 + 0.05)
   400→   400→            }
   401→   401→          }
   402→   402→        }
   403→   403→        // Key reveals.
   404→   404→        for (const k of this.keys) {
   405→   405→          if (k.found) continue
   406→   406→          const kd = Math.hypot(k.x - p.ox, k.y - p.oy)
   407→   407→          if (kd <= p.r && kd > rPrev - 0.5) {
   408→   408→            k.revealT = this.time
   409→   409→            if (p.owner === 0 && this.phase === 'playing') {
   410→   410→              this.audio.jingle(this.panOf(k.x), 0.16)
   411→   411→            }
   412→   412→          }
   413→   413→        }
   414→   414→        // Monster / survivor-ghost reveal (only survivor pings screech).
   415→   415→        const m = this.monster
   416→   416→        if (m && p.owner === 0) {
   417→   417→          const md = Math.hypot(m.x - p.ox, m.y - p.oy)
   418→   418→          if (md <= p.r && md > rPrev - 0.5) {
   419→   419→            m.revealT = this.time
   420→   420→            if (this.phase === 'playing' && this.mode !== 'hunter' && this.screechGate <= 0) {
   421→   421→              this.audio.screech(this.panOf(m.x), clamp(0.6 - md / 1400, 0.22, 0.6))
   422→   422→              this.screechGate = 1.5
   423→   423→              if (this.mode === 'solo' && m.state !== 'chase' && this.time > GRACE) {
   424→   424→                m.pendingAt = this.time + 0.3
   425→   425→                m.pendingX = this.player.x
   426→   426→                m.pendingY = this.player.y
   427→   427→              }
   428→   428→            }
   429→   429→          }
   430→   430→        }
   431→   431→      }
   432→   432→      if (p.r >= p.maxR) p.dead = true
   433→   433→    }
   434→   434→    this.pings = this.pings.filter((p) => !p.dead)
   435→   435→  }
   436→   436→
   437→   437→  private updatePlayer(dt: number): void {
   438→   438→    let { x, y } = this.input
   439→   439→    const l = Math.hypot(x, y)
   440→   440→    if (l > 1) {
   441→   441→      x /= l
   442→   442→      y /= l
   443→   443→    }
   444→   444→    const moving = l > 0.08
   445→   445→    const sp = PLAYER_SPEED
   446→   446→    const nx = this.player.x + x * sp * dt
   447→   447→    if (!this.hitsWall(nx, this.player.y)) this.player.x = nx
   448→   448→    const ny = this.player.y + y * sp * dt
   449→   449→    if (!this.hitsWall(this.player.x, ny)) this.player.y = ny
   450→   450→
   451→   451→    if (moving) {
   452→   452→      this.stepT -= dt
   453→   453→      if (this.stepT <= 0) {
   454→   454→        this.stepT = this.mode === 'hunter' ? 0.3 : 0.34
   455→   455→        if (this.mode === 'hunter') this.audio.monsterStep(0, 0.05)
   456→   456→        else this.audio.step(0.045)
   457→   457→      }
   458→   458→    }
   459→   459→  }
   460→   460→
   461→   461→  private updateKeys(dt: number): void {
   462→   462→    for (const k of this.keys) {
   463→   463→      if (k.found) continue
   464→   464→      const d = Math.hypot(this.player.x - k.x, this.player.y - k.y)
   465→   465→      // Faint jingle hints when a key is close by.
   466→   466→      k.jT -= dt
   467→   467→      if (d < 300 && k.jT <= 0) {
   468→   468→        k.jT = 3.1 + Math.random() * 1.6
   469→   469→        this.audio.jingle(this.panOf(k.x), 0.05 + 0.1 * this.volByDist(d, 300))
   470→   470→      }
   471→   471→      if (d < 24) {
   472→   472→        k.found = true
   473→   473→        this.keysFound++
   474→   474→        this.audio.chime(this.panOf(k.x), 0.5)
   475→   475→        const counts = this.counts()
   476→   476→        this.hooks.onKeys?.(counts)
   477→   477→        if (this.mode === 'solo') {
   478→   478→          if (this.keysFound >= 3) {
   479→   479→            const d0 = this.doorState[0]
   480→   480→            if (d0) d0.unlocked = true
   481→   481→            this.audio.unlock()
   482→   482→            this.hooks.onToast?.('THE DOOR IS OPEN')
   483→   483→            const m = this.monster
   484→   484→            if (m && m.state === 'dormant') m.state = 'wander'
   485→   485→          }
   486→   486→        } else if (k.color !== 'white' && counts[k.color] >= 3) {
   487→   487→          this.hooks.onToast?.(`ALL ${k.color.toUpperCase()} KEYS — THAT DOOR CAN BE UNLOCKED`)
   488→   488→        }
   489→   489→      }
   490→   490→    }
   491→   491→  }

   493→   493→  private counts(): KeyCounts {
   494→   494→    let crimson = 0
   495→   495→    let jade = 0
   496→   496→    for (const k of this.keys) {
   497→   497→      if (!k.found) continue
   498→   498→      if (k.color === 'crimson') crimson++
   499→   499→      else if (k.color === 'jade') jade++
   500→   500→    }
   501→   501→    return { total: this.keysFound, crimson, jade }
   502→   502→  }

   507→   507→    return n

   510→   510→  private updateDoors(dt: number): void {
   511→   511→    const map = this.map
   512→   512→    if (!map) return
   513→   513→    let hum = 0
   514→   514→    for (let i = 0; i < this.doorState.length; i++) {
   515→   515→      const ds = this.doorState[i]
   516→   516→      const d = Math.hypot(this.player.x - ds.spec.midX, this.player.y - ds.spec.midY)
   517→   517→
   518→   518→      if (ds.unlocked) {
   519→   519→        // Open door: hum + beacon pings, and the way out.
   520→   520→        hum = Math.max(hum, this.volByDist(d, 300))
   521→   521→        ds.beaconT -= dt
   522→   522→        if (ds.beaconT <= 0) {
   523→   523→          ds.beaconT = 2.4
   524→   524→          this.pings.push(this.makePing(ds.spec.midX, ds.spec.midY, 1, 170, 320))
   525→   525→          if (d < 620)
   526→   526→            this.audio.doorBlip(this.panOf(ds.spec.midX), 0.05 + 0.14 * this.volByDist(d, 620))
   527→   527→        }
   528→   528→        if (this.phase === 'playing' && d < (this.mode === 'solo' ? 42 : ESCAPE_R)) {
   529→   529→          this.finishEscape(i)
   530→   530→        }
   531→   531→        continue
   532→   532→      }
   533→   533→
   534→   534→      if (this.mode !== 'host') continue
   535→   535→
   536→   536→      // --- Hunt lock channel: all 3 keys of this door's color, then hold
   537→   537→      //     still beside it for three full seconds. Step away and the lock
   538→   538→      //     slams back shut. ---
   539→   539→      const ready = this.keysOf(ds.spec.color) >= 3
   540→   540→      if (ready && d < UNLOCK_R) {
   541→   541→        if (!ds.channeling) {
   542→   542→          ds.channeling = true
   543→   543→          ds.tickT = 0
   544→   544→          this.hooks.onToast?.('UNLOCKING — HOLD STILL')
   545→   545→        }
   546→   546→        ds.progress += dt / UNLOCK_TIME
   547→   547→        ds.tickT -= dt
   548→   548→        if (ds.tickT <= 0) {
   549→   549→          ds.tickT = 0.75
   550→   550→          this.audio.doorBlip(0, 0.1 + 0.22 * ds.progress)
   551→   551→        }
   552→   552→        const step = Math.floor(ds.progress * 8)
   553→   553→        if (step !== ds.lastStep) {
   554→   554→          ds.lastStep = step
   555→   555→          this.emitDoor(i, ds)
   556→   556→        }
   557→   557→        if (ds.progress >= 1) {
   558→   558→          ds.progress = 1
   559→   559→          ds.unlocked = true
   560→   560→          ds.channeling = false
   561→   561→          this.audio.unlock(this.panOf(ds.spec.midX))
   562→   562→          this.hooks.onToast?.(`THE ${ds.spec.color.toUpperCase()} DOOR IS OPEN — GO`)
   563→   563→          this.emitDoor(i, ds)
   564→   564→        }
   565→   565→      } else {
   566→   566→        if (ds.channeling && ds.progress > 0.08) {
   567→   567→          this.hooks.onToast?.('UNLOCK INTERRUPTED')
   568→   568→          this.audio.thud(this.panOf(ds.spec.midX), 0.3)
   569→   569→        }
   570→   570→        if (ds.channeling && ds.progress > 0) {
   571→   571→          ds.progress = 0
   572→   572→          ds.lastStep = -1
   573→   573→          this.emitDoor(i, ds)
   574→   574→        }
   575→   575→        ds.channeling = false
   576→   576→      }
   577→   577→
   578→   578→      // A lock you can work calls to you — a faint beacon marks the door.
   579→   579→      if (ready) {
   580→   580→        ds.beaconT -= dt
   581→   581→        if (ds.beaconT <= 0) {
   582→   582→          ds.beaconT = 3.4
   583→   583→          this.pings.push(this.makePing(ds.spec.midX, ds.spec.midY, 1, 190, 300))
   584→   584→          if (d < 520)
   585→   585→            this.audio.doorBlip(this.panOf(ds.spec.midX), 0.03 + 0.09 * this.volByDist(d, 520))
   586→   586→        }
   587→   587→      }
   588→   588→    }
   589→   589→    this.audio.setExitHum(hum)
   590→   590→  }

   602→   602→      keysFound: this.keysFound,
   603→   603→      pings: this.pingsUsed,
   604→   604→      door: this.doorState[doorIndex]?.spec.color,
   605→   605→    })
   606→   606→  }

   608→   608→  private emitDoor(i: number, ds: DoorState): void {
   609→   609→    if (ds.spec.color === 'white') return // solo doors never go over the wire
   610→   610→    const progress = Math.min(1, Math.round(ds.progress * 100) / 100)
   611→   611→    this.hooks.onEmit?.({
   612→   612→      t: 'door',
   613→   613→      i: i as 0 | 1,
   614→   614→      color: ds.spec.color,
   615→   615→      progress,
   616→   616→      unlocked: ds.unlocked,
   617→   617→    })

   620→   620→  private updateDanger(dt: number): void {
   621→   621→    const m = this.monster
   622→   622→    if (!m) return
   623→   623→    if (this.time > GRACE) {
   624→   624→      if (!this.graceAnnounced) {
   625→   625→        this.graceAnnounced = true
   626→   626→        this.hooks.onToast?.(
   627→   627→          this.mode === 'solo'
   628→   628→            ? 'SOMETHING IS AWAKE'
   629→   629→            : this.mode === 'hunter'
   630→   630→              ? 'THE PREY IS SOMEWHERE — IT PINGS'
   631→   631→              : 'IT IS SOMEWHERE — LISTENING',
   632→   632→        )
   633→   633→        this.audio.growl(this.panOf(m.x), 0.5)
   634→   634→      }
   635→   635→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   636→   636→      let target = clamp(1 - d / 340, 0, 1)
   637→   637→      if (m.state !== 'chase') target *= 0.6
   638→   638→      this.danger += (target - this.danger) * Math.min(1, dt * 4)
   639→   639→      if (this.danger > 0.04) {
   640→   640→        this.hbT -= dt
   641→   641→        if (this.hbT <= 0) {
   642→   642→          this.hbT = 1.15 - 0.68 * this.danger
   643→   643→          this.audio.heartbeat(0.1 + 0.45 * this.danger)
   644→   644→        }
   645→   645→      }
   646→   646→      if (m.state === 'chase') this.shake = Math.max(this.shake, this.danger * 0.55)
   647→   647→    }
   648→   648→    this.audio.setDanger(this.danger)
   649→   649→  }

   653→   653→  private updateMonster(dt: number): void {
   654→   654→    const map = this.map
   655→   655→    const m = this.monster
   656→   656→    if (!map || !m) return
   657→   657→    if (m.state === 'dormant') {
   658→   658→      if (this.time > GRACE) m.state = 'wander'
   659→   659→      return
   660→   660→    }
   661→   661→
   662→   662→    const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   663→   663→    const mc = worldToCell(map, m.x, m.y)
   664→   664→
   665→   665→    // Sense the player up close — no ping needed.
   666→   666→    if (d < SENSE_R && m.state !== 'chase') {
   667→   667→      m.state = 'chase'
   668→   668→      m.path = []
   669→   669→      m.repath = 0
   670→   670→      m.lost = 0
   671→   671→      if (this.screechGate <= 0) {
   672→   672→        this.audio.screech(this.panOf(m.x), 0.65)
   673→   673→        this.screechGate = 1.2
   674→   674→      }
   675→   675→      if (!this.chaseAnnounced) {
   676→   676→        this.chaseAnnounced = true
   677→   677→        this.hooks.onToast?.('RUN')
   678→   678→      }
   679→   679→    }
   680→   680→
   681→   681→    // A ping it heard (delayed reaction).
   682→   682→    if (m.pendingAt > 0 && this.time >= m.pendingAt) {
   683→   683→      m.pendingAt = 0
   684→   684→      if (m.state !== 'chase') {
   685→   685→        const cell = nearestOpenCell(map, ...this.cellOfPoint(m.pendingX, m.pendingY))
   686→   686→        m.state = 'investigate'
   687→   687→        m.target = cell
   688→   688→        m.path = bfsPath(map, mc.c, mc.r, cell.c, cell.r)
   689→   689→        m.pathI = 0
   690→   690→      }
   691→   691→    }
   692→   692→
   693→   693→    let speed = 60
   694→   694→    if (m.state === 'chase') {
   695→   695→      speed = 148 + this.keysFound * 8
   696→   696→      m.lastPX = this.player.x
   697→   697→      m.lastPY = this.player.y
   698→   698→      if (d > LOSE_R) {
   699→   699→        m.lost += dt
   700→   700→        if (m.lost > 2.2) {
   701→   701→          m.state = 'investigate'
   702→   702→          m.target = nearestOpenCell(map, ...this.cellOfPoint(m.lastPX, m.lastPY))
   703→   703→          m.path = bfsPath(map, mc.c, mc.r, m.target.c, m.target.r)
   704→   704→          m.pathI = 0
   705→   705→        }
   706→   706→      } else {
   707→   707→        m.lost = 0
   708→   708→      }
   709→   709→      m.repath -= dt
   710→   710→      const pc = worldToCell(map, this.player.x, this.player.y)
   711→   711→      if (m.repath <= 0) {
   712→   712→        m.repath = 0.32
   713→   713→        m.path = bfsPath(map, mc.c, mc.r, pc.c, pc.r)
   714→   714→        m.pathI = 0
   715→   715→      }
   716→   716→      m.clickT -= dt
   717→   717→      if (m.clickT <= 0) {
   718→   718→        m.clickT = 2.6
   719→   719→        this.monsterClick(m)
   720→   720→      }
   721→   721→      m.stepT -= dt
   722→   722→      if (m.stepT <= 0 && d < 460) {
   723→   723→        m.stepT = 0.27
   724→   724→        this.audio.monsterStep(this.panOf(m.x), 0.1 + 0.3 * this.volByDist(d, 460))
   725→   725→      }
   726→   726→    } else if (m.state === 'investigate') {
   727→   727→      speed = 98
   728→   728→      if (!m.target || m.pathI >= m.path.length) {
   729→   729→        m.state = 'wander'
   730→   730→        m.target = null
   731→   731→        m.path = []
   732→   732→      }
   733→   733→      m.clickT -= dt
   734→   734→      if (m.clickT <= 0) {
   735→   735→        m.clickT = 4.5 + Math.random() * 2.5
   736→   736→        this.monsterClick(m)
   737→   737→      }
   738→   738→    } else {
   739→   739→      // wander — it drifts through the dark, but its instinct pulls it
   740→   740→      // toward your region of the maze (biased random walk).
   741→   741→      speed = 60
   742→   742→      if (m.pathI >= m.path.length) {
   743→   743→        const cands: Cell[] = []
   744→   744→        for (let i = 0; i < 8; i++) {
   745→   745→          const c = clamp(mc.c + Math.floor((Math.random() - 0.5) * 22), 1, map.cols - 2)
   746→   746→          const r = clamp(mc.r + Math.floor((Math.random() - 0.5) * 16), 1, map.rows - 2)
   747→   747→          const cell = nearestOpenCell(map, c, r)
   748→   748→          if (Math.hypot(cell.c - mc.c, cell.r - mc.r) >= 3) cands.push(cell)
   749→   749→        }
   750→   750→        if (cands.length > 0) {
   751→   751→          let pick = cands[Math.floor(Math.random() * cands.length)]
   752→   752→          if (Math.random() < 0.6) {
   753→   753→            const pc = worldToCell(map, this.player.x, this.player.y)
   754→   754→            for (const c of cands) {
   755→   755→              if (
   756→   756→                Math.hypot(c.c - pc.c, c.r - pc.r) <
   757→   757→                Math.hypot(pick.c - pc.c, pick.r - pc.r)
   758→   758→              ) {
   759→   759→                pick = c
   760→   760→              }
   761→   761→            }
   762→   762→          }
   763→   763→          m.path = bfsPath(map, mc.c, mc.r, pick.c, pick.r)
   764→   764→          m.pathI = 0
   765→   765→        }
   766→   766→      }
   767→   767→      m.growlT -= dt
   768→   768→      if (m.growlT <= 0) {
   769→   769→        m.growlT = 3.5 + Math.random() * 4.5
   770→   770→        this.audio.growl(this.panOf(m.x), 0.06 + 0.4 * this.volByDist(d, 560))
   771→   771→      }
   772→   772→      m.clickT -= dt
   773→   773→      if (m.clickT <= 0) {
   774→   774→        m.clickT = 6 + Math.random() * 3
   775→   775→        this.monsterClick(m)
   776→   776→      }
   777→   777→    }
   778→   778→
   779→   779→    // Follow path.
   780→   780→    if (m.pathI < m.path.length) {
   781→   781→      const wp = m.path[m.pathI]
   782→   782→      const dx = wp.x - m.x
   783→   783→      const dy = wp.y - m.y
   784→   784→      const dl = Math.hypot(dx, dy)
   785→   785→      if (dl < 6) {
   786→   786→        m.pathI++
   787→   787→      } else {
   788→   788→        m.x += (dx / dl) * speed * dt
   789→   789→        m.y += (dy / dl) * speed * dt
   790→   790→      }
   791→   791→    }
   792→   792→
   793→   793→    // Catch.
   794→   794→    this.checkCatch(d)
   795→   795→  }
   796→   796→
   797→   797→  private checkCatch(d: number): void {
   798→   798→    if (d >= CATCH_R || this.phase !== 'playing') return
   799→   799→    const m = this.monster
   800→   800→    this.deathAt = this.time
   801→   801→    this.phase = 'dead'
   802→   802→    this.shake = 1.4
   803→   803→    this.audio.setDanger(0)
   804→   804→    this.audio.setExitHum(0)
   805→   805→    this.audio.death()
   806→   806→    if (m) m.revealT = this.time
   807→   807→    this.hooks.onEnd?.({
   808→   808→      outcome: 'caught',
   809→   809→      timeMs: Math.round(this.time * 1000),
   810→   810→      keysFound: this.keysFound,
   811→   811→      pings: this.pingsUsed,
   812→   812→    })
   813→   813→  }
