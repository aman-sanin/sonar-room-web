     1→// The Sonar Room — core game engine.
     2→// You are blind: the world only exists for the ~1.5s a ping takes to fade.
     3→// Pings expand as wavefronts; where they touch wall segments the wireframe
     4→// blinks into existence. Sound is spatialized. The thing in the dark hears
     5→// every ping you make.
     6→
     7→import {
     8→  GameMap,
     9→  Vec,
    10→  Cell,
    11→  generateMap,
    12→  bfsPath,
    13→  cellCenter,
    14→  isSolid,
    15→  nearestOpenCell,
    16→  segDist,
    17→  worldToCell,
    18→} from './maze'
    19→import { SoundEngine } from './audio'
    20→
    21→export type EnginePhase = 'idle' | 'playing' | 'dead' | 'win'
    22→
    23→export interface RunResult {
    24→  outcome: 'escaped' | 'caught'
    25→  timeMs: number
    26→  keysFound: number
    27→  pings: number
    28→}
    29→
    30→export interface EngineHooks {
    31→  onKeys?: (n: number) => void
    32→  onToast?: (msg: string) => void
    33→  onEnd?: (r: RunResult) => void
    34→  /** Multiplayer: events the engine asks the component to relay to the peer. */
    35→  onEmit?: (e: HuntEmit) => void
    36→}
    37→
    38→/** solo = vs AI · host = survivor, monster is remote · hunter = you are the monster. */
    39→export type EngineMode = 'solo' | 'host' | 'hunter'
    40→
    41→export type HuntEmit =
    42→  | { t: 'ping'; x: number; y: number }
    43→  | { t: 'click'; x: number; y: number }
    44→
    45→interface Ping {
    46→  ox: number
    47→  oy: number
    48→  r: number
    49→  maxR: number
    50→  speed: number
    51→  owner: 0 | 1 // 0 = player, 1 = monster/door (dimmer, silent reveal)
    52→  hits: Uint8Array
    53→  doorHit: boolean
    54→  jag: number[]
    55→  dead: boolean
    56→}
    57→
    58→type MonsterState = 'dormant' | 'wander' | 'investigate' | 'chase'
    59→
    60→interface Monster {
    61→  x: number
    62→  y: number
    63→  state: MonsterState
    64→  path: Vec[]
    65→  pathI: number
    66→  target: Cell | null
    67→  repath: number
    68→  lost: number
    69→  growlT: number
    70→  clickT: number
    71→  stepT: number
    72→  pendingAt: number
    73→  pendingX: number
    74→  pendingY: number
    75→  revealT: number
    76→  lastPX: number
    77→  lastPY: number
    78→  /** Multiplayer: latest peer-reported position to ease toward. */
    79→  netX: number
    80→  netY: number
    81→  netMoving: boolean
    82→}
    83→
    84→interface KeyItem {
    85→  x: number
    86→  y: number
    87→  found: boolean
    88→  revealT: number
    89→  jT: number

    91→
    92→const PING_MAX_R = 640
    93→const PING_SPEED = 500
    94→const PING_COOLDOWN = 1.05
    95→const PLAYER_SPEED = 152
    96→const PLAYER_R = 9
    97→const SENSE_R = 150 // monster senses you this close, ping or not
    98→const LOSE_R = 310 // break this far for long enough and it loses you
    99→const HEAR_R = 920 // it hears any ping made within this range
   100→const CATCH_R = 20
   101→const GRACE = 8 // seconds before the thing wakes up
   102→
   103→const clamp = (v: number, a: number, b: number): number => Math.max(a, Math.min(b, v))
   104→const expFade = (age: number, tau: number): number => Math.exp(-age / tau)
   105→
   106→export class SonarEngine {
   107→  map: GameMap | null = null
   108→  phase: EnginePhase = 'idle'
   109→  time = 0
   110→  cooldown = 0
   111→  pingsUsed = 0
   112→  keysFound = 0
   113→  unlocked = false
   114→  danger = 0
   115→  input = { x: 0, y: 0 }
   116→  mode: EngineMode = 'solo'
   117→
   118→  private player = { x: 0, y: 0 }
   119→  private cam = { x: 0, y: 0 }
   120→  private monster: Monster | null = null
   121→  private keys: KeyItem[] = []
   122→  private pings: Ping[] = []
   123→  private revealT: Float32Array = new Float32Array(0)
   124→  private revealS: Float32Array = new Float32Array(0)
   125→  private doorRevealT = -99
   126→  private shake = 0
   127→  private hbT = 0
   128→  private stepT = 0
   129→  private thudGate = 0
   130→  private screechGate = 0
   131→  private beaconT = 0
   132→  private graceAnnounced = false
   133→  private chaseAnnounced = false
   134→  private deathAt = 0
   135→  private winAt = 0
   136→  private seed = 1
   137→
   138→  constructor(
   139→    private audio: SoundEngine,
   140→    private hooks: EngineHooks = {},
   141→  ) {}
   142→
   143→  private panOf(x: number): number {
   144→    return clamp((x - this.player.x) / 300, -1, 1)
   145→  }
   146→
   147→  private volByDist(d: number, range: number): number {
   148→    return clamp(1 - d / range, 0, 1)
   149→  }
   150→
   151→  start(seed = Math.floor(Math.random() * 1e9)): void {
   152→    this.seed = seed
   153→    this.map = generateMap(seed)
   154→    const n = this.map.edges.length
   155→    this.revealT = new Float32Array(n).fill(-99)
   156→    this.revealS = new Float32Array(n)
   157→    // In hunter mode the LOCAL body is the monster; the "monster" slot
   158→    // holds the remote survivor ghost. Audio panning stays relative to
   159→    // the local body either way.
   160→    this.player = this.mode === 'hunter' ? { ...this.map.monsterStart } : { ...this.map.start }
   161→    this.cam = { ...this.player }
   162→    this.keys = this.map.keys.map((k) => ({ x: k.x, y: k.y, found: false, revealT: -99, jT: 1 + Math.random() * 2 }))
   163→    const m: Monster = {
   164→      x: this.mode === 'hunter' ? this.map.start.x : this.map.monsterStart.x,
   165→      y: this.mode === 'hunter' ? this.map.start.y : this.map.monsterStart.y,
   166→      state: 'dormant',
   167→      path: [],
   168→      pathI: 0,
   169→      target: null,
   170→      repath: 0,
   171→      lost: 0,
   172→      growlT: 2 + Math.random() * 3,
   173→      clickT: 3 + Math.random() * 3,
   174→      stepT: 0,
   175→      pendingAt: 0,
   176→      pendingX: 0,
   177→      pendingY: 0,
   178→      revealT: -99,
   179→      lastPX: 0,
   180→      lastPY: 0,
   181→      netX: 0,
   182→      netY: 0,
   183→      netMoving: false,
   184→    }
   185→    m.netX = m.x
   186→    m.netY = m.y
   187→    this.monster = m
   188→    this.pings = []
   189→    this.time = 0
   190→    this.cooldown = 0
   191→    this.pingsUsed = 0
   192→    this.keysFound = 0
   193→    this.unlocked = false
   194→    this.danger = 0
   195→    this.shake = 0
   196→    this.doorRevealT = -99
   197→    this.thudGate = 0
   198→    this.screechGate = 0
   199→    this.beaconT = 0
   200→    this.graceAnnounced = false
   201→    this.chaseAnnounced = false
   202→    this.phase = 'playing'
   203→    this.audio.setDanger(0)
   204→    this.audio.setExitHum(0)
   205→    this.hooks.onKeys?.(0)
   206→    this.hooks.onToast?.(
   207→      this.mode === 'hunter' ? 'SOMETHING FLESH IS IN THE DARK — LISTEN' : 'FIND THE KEYS — QUIETLY',
   208→    )
   209→  }
   210→
   211→  /** Emit a sonar ping — or, as the hunter, an echolocation click. Costs noise. */
   212→  ping(): boolean {
   213→    if (this.phase !== 'playing' || this.cooldown > 0 || !this.map) return false
   214→    this.cooldown = PING_COOLDOWN
   215→    this.pingsUsed++
   216→    if (this.mode === 'hunter') {
   217→      this.pings.push(this.makePing(this.player.x, this.player.y, 1, 430, 440))
   218→      this.audio.monsterClick(0, 0.32)
   219→      this.hooks.onEmit?.({ t: 'click', x: this.player.x, y: this.player.y })
   220→      return true
   221→    }
   222→    this.pings.push(this.makePing(this.player.x, this.player.y, 0, PING_MAX_R, PING_SPEED))
   223→    this.audio.ping()
   224→    if (this.mode === 'host') {
   225→      this.hooks.onEmit?.({ t: 'ping', x: this.player.x, y: this.player.y })
   226→    }
   227→    const m = this.monster
   228→    if (this.mode === 'solo' && m && m.state !== 'chase' && this.time > GRACE) {
   229→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   230→      if (d < HEAR_R) {
   231→        m.pendingAt = this.time + 0.35
   232→        m.pendingX = this.player.x
   233→        m.pendingY = this.player.y
   234→      }
   235→    }
   236→    return true
   237→  }
   238→
   239→  private makePing(ox: number, oy: number, owner: 0 | 1, maxR: number, speed: number): Ping {
   240→    const jag: number[] = []
   241→    for (let i = 0; i < 48; i++) jag.push((Math.random() - 0.5) * 2)
   242→    return {
   243→      ox,
   244→      oy,
   245→      r: 2,
   246→      maxR,
   247→      speed,
   248→      owner,
   249→      hits: new Uint8Array(this.map ? this.map.edges.length : 0),
   250→      doorHit: false,
   251→      jag,
   252→      dead: false,
   253→    }
   254→  }
   255→
   256→  // ------------------------------------------------------------- collision
   257→
   258→  private hitsWall(x: number, y: number): boolean {
   259→    const map = this.map
   260→    if (!map) return false
   261→    const c0 = Math.floor((x - PLAYER_R) / map.cell)
   262→    const c1 = Math.floor((x + PLAYER_R) / map.cell)
   263→    const r0 = Math.floor((y - PLAYER_R) / map.cell)
   264→    const r1 = Math.floor((y + PLAYER_R) / map.cell)
   265→    for (let r = r0; r <= r1; r++) {
   266→      for (let c = c0; c <= c1; c++) {
   267→        if (!isSolid(map, c, r)) continue
   268→        const rx = c * map.cell
   269→        const ry = r * map.cell
   270→        const nx = clamp(x, rx, rx + map.cell)
   271→        const ny = clamp(y, ry, ry + map.cell)
   272→        if (Math.hypot(x - nx, y - ny) < PLAYER_R) return true
   273→      }
   274→    }
   275→    return false
   276→  }

   278→  // ---------------------------------------------------------------- update
   279→
   280→  update(dt: number): void {
   281→    this.time += dt
   282→    this.cooldown = Math.max(0, this.cooldown - dt)
   283→    this.thudGate -= dt
   284→    this.screechGate -= dt
   285→    this.shake = Math.max(0, this.shake - dt * 1.6)
   286→
   287→    this.updatePings(dt)
   288→
   289→    if (this.phase === 'playing') {
   290→      this.updatePlayer(dt)
   291→      if (this.mode === 'hunter') {
   292→        this.updateGhost(dt)
   293→        this.updateDanger(dt)
   294→      } else if (this.mode === 'host') {
   295→        this.updateRemoteMonster(dt)
   296→        this.updateKeys(dt)
   297→        this.updateExit(dt)
   298→        this.updateDanger(dt)
   299→      } else {
   300→        this.updateMonster(dt)
   301→        this.updateKeys(dt)
   302→        this.updateExit(dt)
   303→        this.updateDanger(dt)
   304→      }
   305→    } else {
   306→      this.danger = Math.max(0, this.danger - dt * 0.8)
   307→      this.audio.setDanger(this.danger)
   308→    }
   309→
   310→    // Camera follows.
   311→    const k = Math.min(1, dt * 4.5)
   312→    this.cam.x += (this.player.x - this.cam.x) * k
   313→    this.cam.y += (this.player.y - this.cam.y) * k
   314→  }
   315→
   316→  private updatePings(dt: number): void {
   317→    const map = this.map
   318→    if (!map) return
   319→    for (const p of this.pings) {
   320→      const rPrev = p.r
   321→      p.r = Math.min(p.maxR, p.r + p.speed * dt)
   322→      if (p.owner === 0 || p.owner === 1) {
   323→        // Edge crossings.
   324→        let soundBudget = 2
   325→        for (let i = 0; i < map.edges.length; i++) {
   326→          if (p.hits[i]) continue
   327→          const e = map.edges[i]
   328→          const mx = (e.x1 + e.x2) / 2
   329→          const my = (e.y1 + e.y2) / 2
   330→          // Cheap reject by bounding box.
   331→          const bx = Math.min(e.x1, e.x2) - p.r
   332→          if (p.ox < bx || p.ox > Math.max(e.x1, e.x2) + p.r) {
   333→            if (Math.abs(p.ox - mx) > p.r) continue
   334→          }
   335→          const d = segDist(p.ox, p.oy, e.x1, e.y1, e.x2, e.y2)
   336→          if (d <= p.r && d > rPrev - 0.5) {
   337→            p.hits[i] = 1
   338→            const strength = clamp(1.08 - d / 720, 0.28, 1) * (p.owner === 1 ? 0.5 : 1)
   339→            if (strength > this.revealS[i] * expFade(this.time - this.revealT[i], 0.55)) {
   340→              this.revealS[i] = strength
   341→              this.revealT[i] = this.time
   342→            }
   343→            if (p.owner === 0 && this.phase === 'playing' && soundBudget > 0 && this.thudGate <= 0) {
   344→              const vol = strength * clamp(1.2 - d / 620, 0.2, 1) * 0.85
   345→              this.audio.thud(this.panOf(mx), vol)
   346→              soundBudget--
   347→              this.thudGate = 0.045
   348→            }
   349→          }
   350→        }
   351→        // Door reveal.
   352→        const dd = Math.hypot(map.door.midX - p.ox, map.door.midY - p.oy)
   353→        if (dd <= p.r && dd > rPrev - 0.5) {
   354→          this.doorRevealT = this.time
   355→          if (p.owner === 0 && !p.doorHit && this.phase === 'playing') {
   356→            p.doorHit = true
   357→            this.audio.doorTone(this.panOf(map.door.midX), this.volByDist(dd, 640) * 0.5 + 0.05)
   358→          }
   359→        }
   360→        // Key reveals.
   361→        for (const k of this.keys) {
   362→          if (k.found) continue
   363→          const kd = Math.hypot(k.x - p.ox, k.y - p.oy)
   364→          if (kd <= p.r && kd > rPrev - 0.5) {
   365→            k.revealT = this.time
   366→            if (p.owner === 0 && this.phase === 'playing') {
   367→              this.audio.jingle(this.panOf(k.x), 0.16)
   368→            }
   369→          }
   370→        }
   371→        // Monster / survivor-ghost reveal (only survivor pings screech).
   372→        const m = this.monster
   373→        if (m && p.owner === 0) {
   374→          const md = Math.hypot(m.x - p.ox, m.y - p.oy)
   375→          if (md <= p.r && md > rPrev - 0.5) {
   376→            m.revealT = this.time
   377→            if (this.phase === 'playing' && this.mode !== 'hunter' && this.screechGate <= 0) {
   378→              this.audio.screech(this.panOf(m.x), clamp(0.6 - md / 1400, 0.22, 0.6))
   379→              this.screechGate = 1.5
   380→              if (this.mode === 'solo' && m.state !== 'chase' && this.time > GRACE) {
   381→                m.pendingAt = this.time + 0.3
   382→                m.pendingX = this.player.x
   383→                m.pendingY = this.player.y
   384→              }
   385→            }
   386→          }
   387→        }
   388→      }
   389→      if (p.r >= p.maxR) p.dead = true
   390→    }
   391→    this.pings = this.pings.filter((p) => !p.dead)
   392→  }
   393→
   394→  private updatePlayer(dt: number): void {
   395→    let { x, y } = this.input
   396→    const l = Math.hypot(x, y)
   397→    if (l > 1) {
   398→      x /= l
   399→      y /= l
   400→    }
   401→    const moving = l > 0.08
   402→    const sp = PLAYER_SPEED
   403→    const nx = this.player.x + x * sp * dt
   404→    if (!this.hitsWall(nx, this.player.y)) this.player.x = nx
   405→    const ny = this.player.y + y * sp * dt
   406→    if (!this.hitsWall(this.player.x, ny)) this.player.y = ny
   407→
   408→    if (moving) {
   409→      this.stepT -= dt
   410→      if (this.stepT <= 0) {
   411→        this.stepT = this.mode === 'hunter' ? 0.3 : 0.34
   412→        if (this.mode === 'hunter') this.audio.monsterStep(0, 0.05)
   413→        else this.audio.step(0.045)
   414→      }
   415→    }
   416→  }
   417→
   418→  private updateKeys(dt: number): void {
   419→    for (const k of this.keys) {
   420→      if (k.found) continue
   421→      const d = Math.hypot(this.player.x - k.x, this.player.y - k.y)
   422→      // Faint jingle hints when a key is close by.
   423→      k.jT -= dt
   424→      if (d < 300 && k.jT <= 0) {
   425→        k.jT = 3.1 + Math.random() * 1.6
   426→        this.audio.jingle(this.panOf(k.x), 0.05 + 0.1 * this.volByDist(d, 300))
   427→      }
   428→      if (d < 24) {
   429→        k.found = true
   430→        this.keysFound++
   431→        this.audio.chime(this.panOf(k.x), 0.5)
   432→        this.hooks.onKeys?.(this.keysFound)
   433→        if (this.keysFound >= 3) {
   434→          this.unlocked = true
   435→          this.audio.unlock()
   436→          this.hooks.onToast?.('THE DOOR IS OPEN')
   437→          const m = this.monster
   438→          if (m && m.state === 'dormant') m.state = 'wander'
   439→        }
   440→      }
   441→    }
   442→  }
   443→
   444→  private updateExit(dt: number): void {
   445→    const map = this.map
   446→    if (!map) return
   447→    const d = Math.hypot(this.player.x - map.door.midX, this.player.y - map.door.midY)
   448→    if (this.unlocked) {
   449→      this.audio.setExitHum(this.volByDist(d, 280))
   450→      this.beaconT -= dt
   451→      if (this.beaconT <= 0) {
   452→        this.beaconT = 2.4
   453→        this.pings.push(this.makePing(map.door.midX, map.door.midY, 1, 170, 320))
   454→        if (d < 620) this.audio.doorBlip(this.panOf(map.door.midX), 0.05 + 0.14 * this.volByDist(d, 620))
   455→      }
   456→      if (d < 42) {
   457→        this.winAt = this.time
   458→        this.phase = 'win'
   459→        this.audio.setDanger(0)
   460→        this.audio.setExitHum(0)
   461→        this.audio.win()
   462→        this.hooks.onEnd?.({
   463→          outcome: 'escaped',
   464→          timeMs: Math.round(this.time * 1000),
   465→          keysFound: this.keysFound,
   466→          pings: this.pingsUsed,
   467→        })
   468→      }
   469→    } else {
   470→      this.audio.setExitHum(0)
   471→    }
   472→  }
   473→
   474→  private updateDanger(dt: number): void {
   475→    const m = this.monster
   476→    if (!m) return
   477→    if (this.time > GRACE) {
   478→      if (!this.graceAnnounced) {
   479→        this.graceAnnounced = true
   480→        this.hooks.onToast?.(
   481→          this.mode === 'solo'
   482→            ? 'SOMETHING IS AWAKE'
   483→            : this.mode === 'hunter'
   484→              ? 'THE PREY IS SOMEWHERE — IT PINGS'
   485→              : 'IT IS SOMEWHERE — LISTENING',
   486→        )
   487→        this.audio.growl(this.panOf(m.x), 0.5)
   488→      }
   489→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   490→      let target = clamp(1 - d / 340, 0, 1)
   491→      if (m.state !== 'chase') target *= 0.6
   492→      this.danger += (target - this.danger) * Math.min(1, dt * 4)
   493→      if (this.danger > 0.04) {
   494→        this.hbT -= dt
   495→        if (this.hbT <= 0) {
   496→          this.hbT = 1.15 - 0.68 * this.danger
   497→          this.audio.heartbeat(0.1 + 0.45 * this.danger)
   498→        }
   499→      }
   500→      if (m.state === 'chase') this.shake = Math.max(this.shake, this.danger * 0.55)
   501→    }
   502→    this.audio.setDanger(this.danger)
   503→  }
   504→
   505→  // -------------------------------------------------------------- monster
   506→
   507→  private updateMonster(dt: number): void {
   508→    const map = this.map
   509→    const m = this.monster
   510→    if (!map || !m) return
   511→    if (m.state === 'dormant') {
   512→      if (this.time > GRACE) m.state = 'wander'
   513→      return
   514→    }
   515→
   516→    const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   517→    const mc = worldToCell(map, m.x, m.y)
   518→
   519→    // Sense the player up close — no ping needed.
   520→    if (d < SENSE_R && m.state !== 'chase') {
   521→      m.state = 'chase'
   522→      m.path = []
   523→      m.repath = 0
   524→      m.lost = 0
   525→      if (this.screechGate <= 0) {
   526→        this.audio.screech(this.panOf(m.x), 0.65)
   527→        this.screechGate = 1.2
   528→      }
   529→      if (!this.chaseAnnounced) {
   530→        this.chaseAnnounced = true
   531→        this.hooks.onToast?.('RUN')
   532→      }
   533→    }
   534→
   535→    // A ping it heard (delayed reaction).
   536→    if (m.pendingAt > 0 && this.time >= m.pendingAt) {
   537→      m.pendingAt = 0
   538→      if (m.state !== 'chase') {
   539→        const cell = nearestOpenCell(map, ...this.cellOfPoint(m.pendingX, m.pendingY))
   540→        m.state = 'investigate'
   541→        m.target = cell
   542→        m.path = bfsPath(map, mc.c, mc.r, cell.c, cell.r)
   543→        m.pathI = 0
   544→      }
   545→    }
   546→
   547→    let speed = 60
   548→    if (m.state === 'chase') {
   549→      speed = 148 + this.keysFound * 8
   550→      m.lastPX = this.player.x
   551→      m.lastPY = this.player.y
   552→      if (d > LOSE_R) {
   553→        m.lost += dt
   554→        if (m.lost > 2.2) {
   555→          m.state = 'investigate'
   556→          m.target = nearestOpenCell(map, ...this.cellOfPoint(m.lastPX, m.lastPY))
   557→          m.path = bfsPath(map, mc.c, mc.r, m.target.c, m.target.r)
   558→          m.pathI = 0
   559→        }
   560→      } else {
   561→        m.lost = 0
   562→      }
   563→      m.repath -= dt
   564→      const pc = worldToCell(map, this.player.x, this.player.y)
   565→      if (m.repath <= 0) {
   566→        m.repath = 0.32
   567→        m.path = bfsPath(map, mc.c, mc.r, pc.c, pc.r)
   568→        m.pathI = 0
   569→      }
   570→      m.clickT -= dt
   571→      if (m.clickT <= 0) {
   572→        m.clickT = 2.6
   573→        this.monsterClick(m)
   574→      }
   575→      m.stepT -= dt
   576→      if (m.stepT <= 0 && d < 460) {
   577→        m.stepT = 0.27
   578→        this.audio.monsterStep(this.panOf(m.x), 0.1 + 0.3 * this.volByDist(d, 460))
   579→      }
   580→    } else if (m.state === 'investigate') {
   581→      speed = 98
   582→      if (!m.target || m.pathI >= m.path.length) {
   583→        m.state = 'wander'
   584→        m.target = null
   585→        m.path = []
   586→      }
   587→      m.clickT -= dt
   588→      if (m.clickT <= 0) {
   589→        m.clickT = 4.5 + Math.random() * 2.5
   590→        this.monsterClick(m)
   591→      }
   592→    } else {
   593→      // wander — it drifts through the dark, but its instinct pulls it
   594→      // toward your region of the maze (biased random walk).
   595→      speed = 60
   596→      if (m.pathI >= m.path.length) {
   597→        const cands: Cell[] = []
   598→        for (let i = 0; i < 8; i++) {
   599→          const c = clamp(mc.c + Math.floor((Math.random() - 0.5) * 22), 1, map.cols - 2)
   600→          const r = clamp(mc.r + Math.floor((Math.random() - 0.5) * 16), 1, map.rows - 2)
   601→          const cell = nearestOpenCell(map, c, r)
   602→          if (Math.hypot(cell.c - mc.c, cell.r - mc.r) >= 3) cands.push(cell)
   603→        }
   604→        if (cands.length > 0) {
   605→          let pick = cands[Math.floor(Math.random() * cands.length)]
   606→          if (Math.random() < 0.6) {
   607→            const pc = worldToCell(map, this.player.x, this.player.y)
   608→            for (const c of cands) {
   609→              if (
   610→                Math.hypot(c.c - pc.c, c.r - pc.r) <
   611→                Math.hypot(pick.c - pc.c, pick.r - pc.r)
   612→              ) {
   613→                pick = c
   614→              }
   615→            }
   616→          }
   617→          m.path = bfsPath(map, mc.c, mc.r, pick.c, pick.r)
   618→          m.pathI = 0
   619→        }
   620→      }
   621→      m.growlT -= dt
   622→      if (m.growlT <= 0) {
   623→        m.growlT = 3.5 + Math.random() * 4.5
   624→        this.audio.growl(this.panOf(m.x), 0.06 + 0.4 * this.volByDist(d, 560))
   625→      }
   626→      m.clickT -= dt
   627→      if (m.clickT <= 0) {
   628→        m.clickT = 6 + Math.random() * 3
   629→        this.monsterClick(m)
   630→      }
   631→    }
   632→
   633→    // Follow path.
   634→    if (m.pathI < m.path.length) {
   635→      const wp = m.path[m.pathI]
   636→      const dx = wp.x - m.x
   637→      const dy = wp.y - m.y
   638→      const dl = Math.hypot(dx, dy)
   639→      if (dl < 6) {
   640→        m.pathI++
   641→      } else {
   642→        m.x += (dx / dl) * speed * dt
   643→        m.y += (dy / dl) * speed * dt
   644→      }
   645→    }
   646→
   647→    // Catch.
   648→    this.checkCatch(d)
   649→  }

   651→  private checkCatch(d: number): void {
   652→    if (d >= CATCH_R || this.phase !== 'playing') return
   653→    const m = this.monster
   654→    this.deathAt = this.time
   655→    this.phase = 'dead'
   656→    this.shake = 1.4
   657→    this.audio.setDanger(0)
   658→    this.audio.setExitHum(0)
   659→    this.audio.death()
   660→    if (m) m.revealT = this.time
   661→    this.hooks.onEnd?.({
   662→      outcome: 'caught',
   663→      timeMs: Math.round(this.time * 1000),
   664→      keysFound: this.keysFound,
   665→      pings: this.pingsUsed,
   666→    })
   667→  }
