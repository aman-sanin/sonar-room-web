     1→     1→// The Sonar Room — core game engine.
     2→     2→// You are blind: the world only exists for the ~1.5s a ping takes to fade.

     4→     4→// blinks into existence. Sound is spatialized. The thing in the dark hears

    79→    79→const PING_MAX_R = 640

    81→    81→const PING_COOLDOWN = 1.05
    82→    82→const PLAYER_SPEED = 152
    83→    83→const PLAYER_R = 9
    84→    84→const SENSE_R = 150 // monster senses you this close, ping or not
    85→    85→const LOSE_R = 310 // break this far for long enough and it loses you
    86→    86→const HEAR_R = 920 // it hears any ping made within this range
    87→    87→const CATCH_R = 20
    88→    88→const GRACE = 8 // seconds before the thing wakes up
    89→    89→

    93→    93→export class SonarEngine {
    94→    94→  map: GameMap | null = null
    95→    95→  phase: EnginePhase = 'idle'
    96→    96→  time = 0
    97→    97→  cooldown = 0
    98→    98→  pingsUsed = 0
    99→    99→  keysFound = 0
   100→   100→  unlocked = false
   101→   101→  danger = 0
   102→   102→  input = { x: 0, y: 0 }
   103→   103→
   104→   104→  private player = { x: 0, y: 0 }
   105→   105→  private cam = { x: 0, y: 0 }
   106→   106→  private monster: Monster | null = null
   107→   107→  private keys: KeyItem[] = []
   108→   108→  private pings: Ping[] = []
   109→   109→  private revealT: Float32Array = new Float32Array(0)
   110→   110→  private revealS: Float32Array = new Float32Array(0)
   111→   111→  private doorRevealT = -99
   112→   112→  private shake = 0
   113→   113→  private hbT = 0
   114→   114→  private stepT = 0
   115→   115→  private thudGate = 0
   116→   116→  private screechGate = 0
   117→   117→  private beaconT = 0
   118→   118→  private graceAnnounced = false
   119→   119→  private chaseAnnounced = false
   120→   120→  private deathAt = 0
   121→   121→  private winAt = 0
   122→   122→  private seed = 1
   123→   123→
   124→   124→  constructor(
   125→   125→    private audio: SoundEngine,
   126→   126→    private hooks: EngineHooks = {},
   127→   127→  ) {}
   128→   128→
   129→   129→  private panOf(x: number): number {
   130→   130→    return clamp((x - this.player.x) / 300, -1, 1)
   131→   131→  }
   132→   132→
   133→   133→  private volByDist(d: number, range: number): number {
   134→   134→    return clamp(1 - d / range, 0, 1)
   135→   135→  }
   136→   136→
   137→   137→  start(seed = Math.floor(Math.random() * 1e9)): void {
   138→   138→    this.seed = seed
   139→   139→    this.map = generateMap(seed)
   140→   140→    const n = this.map.edges.length
   141→   141→    this.revealT = new Float32Array(n).fill(-99)
   142→   142→    this.revealS = new Float32Array(n)
   143→   143→    this.player = { ...this.map.start }
   144→   144→    this.cam = { ...this.map.start }
   145→   145→    this.keys = this.map.keys.map((k) => ({ x: k.x, y: k.y, found: false, revealT: -99, jT: 1 + Math.random() * 2 }))
   146→   146→    this.monster = {
   147→   147→      x: this.map.monsterStart.x,
   148→   148→      y: this.map.monsterStart.y,
   149→   149→      state: 'dormant',
   150→   150→      path: [],
   151→   151→      pathI: 0,
   152→   152→      target: null,
   153→   153→      repath: 0,
   154→   154→      lost: 0,
   155→   155→      growlT: 2 + Math.random() * 3,
   156→   156→      clickT: 3 + Math.random() * 3,
   157→   157→      stepT: 0,
   158→   158→      pendingAt: 0,
   159→   159→      pendingX: 0,
   160→   160→      pendingY: 0,
   161→   161→      revealT: -99,
   162→   162→      lastPX: 0,
   163→   163→      lastPY: 0,
   164→   164→    }
   165→   165→    this.pings = []
   166→   166→    this.time = 0
   167→   167→    this.cooldown = 0
   168→   168→    this.pingsUsed = 0
   169→   169→    this.keysFound = 0
   170→   170→    this.unlocked = false
   171→   171→    this.danger = 0
   172→   172→    this.shake = 0
   173→   173→    this.doorRevealT = -99
   174→   174→    this.thudGate = 0
   175→   175→    this.screechGate = 0
   176→   176→    this.beaconT = 0
   177→   177→    this.graceAnnounced = false
   178→   178→    this.chaseAnnounced = false
   179→   179→    this.phase = 'playing'
   180→   180→    this.audio.setDanger(0)
   181→   181→    this.audio.setExitHum(0)
   182→   182→    this.hooks.onKeys?.(0)
   183→   183→    this.hooks.onToast?.('FIND THE KEYS — QUIETLY')
   184→   184→  }
   185→   185→
   186→   186→  /** Emit a sonar ping from the player. Costs noise. */
   187→   187→  ping(): boolean {
   188→   188→    if (this.phase !== 'playing' || this.cooldown > 0 || !this.map) return false
   189→   189→    this.cooldown = PING_COOLDOWN
   190→   190→    this.pingsUsed++
   191→   191→    this.pings.push(this.makePing(this.player.x, this.player.y, 0, PING_MAX_R, PING_SPEED))
   192→   192→    this.audio.ping()
   193→   193→    const m = this.monster
   194→   194→    if (m && m.state !== 'chase' && this.time > GRACE) {
   195→   195→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   196→   196→      if (d < HEAR_R) {
   197→   197→        m.pendingAt = this.time + 0.35
   198→   198→        m.pendingX = this.player.x
   199→   199→        m.pendingY = this.player.y
   200→   200→      }
   201→   201→    }
   202→   202→    return true
   203→   203→  }
   204→   204→
   205→   205→  private makePing(ox: number, oy: number, owner: 0 | 1, maxR: number, speed: number): Ping {
   206→   206→    const jag: number[] = []
   207→   207→    for (let i = 0; i < 48; i++) jag.push((Math.random() - 0.5) * 2)
   208→   208→    return {
   209→   209→      ox,
   210→   210→      oy,
   211→   211→      r: 2,
   212→   212→      maxR,
   213→   213→      speed,
   214→   214→      owner,
   215→   215→      hits: new Uint8Array(this.map ? this.map.edges.length : 0),
   216→   216→      doorHit: false,
   217→   217→      jag,
   218→   218→      dead: false,
   219→   219→    }
   220→   220→  }
   221→   221→
   222→   222→  // ------------------------------------------------------------- collision
   223→   223→
   224→   224→  private hitsWall(x: number, y: number): boolean {
   225→   225→    const map = this.map
   226→   226→    if (!map) return false
   227→   227→    const c0 = Math.floor((x - PLAYER_R) / map.cell)
   228→   228→    const c1 = Math.floor((x + PLAYER_R) / map.cell)
   229→   229→    const r0 = Math.floor((y - PLAYER_R) / map.cell)
   230→   230→    const r1 = Math.floor((y + PLAYER_R) / map.cell)
   231→   231→    for (let r = r0; r <= r1; r++) {
   232→   232→      for (let c = c0; c <= c1; c++) {
   233→   233→        if (!isSolid(map, c, r)) continue
   234→   234→        const rx = c * map.cell
   235→   235→        const ry = r * map.cell
   236→   236→        const nx = clamp(x, rx, rx + map.cell)
   237→   237→        const ny = clamp(y, ry, ry + map.cell)
   238→   238→        if (Math.hypot(x - nx, y - ny) < PLAYER_R) return true
   239→   239→      }
   240→   240→    }
   241→   241→    return false
   242→   242→  }

   244→   244→  // ---------------------------------------------------------------- update
   245→   245→
   246→   246→  update(dt: number): void {
   247→   247→    this.time += dt
   248→   248→    this.cooldown = Math.max(0, this.cooldown - dt)
   249→   249→    this.thudGate -= dt
   250→   250→    this.screechGate -= dt
   251→   251→    this.shake = Math.max(0, this.shake - dt * 1.6)
   252→   252→
   253→   253→    this.updatePings(dt)
   254→   254→
   255→   255→    if (this.phase === 'playing') {
   256→   256→      this.updatePlayer(dt)
   257→   257→      this.updateMonster(dt)
   258→   258→      this.updateKeys(dt)
   259→   259→      this.updateExit(dt)
   260→   260→      this.updateDanger(dt)
   261→   261→    } else {
   262→   262→      this.danger = Math.max(0, this.danger - dt * 0.8)
   263→   263→      this.audio.setDanger(this.danger)
   264→   264→    }
   265→   265→
   266→   266→    // Camera follows.
   267→   267→    const k = Math.min(1, dt * 4.5)
   268→   268→    this.cam.x += (this.player.x - this.cam.x) * k
   269→   269→    this.cam.y += (this.player.y - this.cam.y) * k
   270→   270→  }
   271→   271→
   272→   272→  private updatePings(dt: number): void {
   273→   273→    const map = this.map
   274→   274→    if (!map) return
   275→   275→    for (const p of this.pings) {
   276→   276→      const rPrev = p.r
   277→   277→      p.r = Math.min(p.maxR, p.r + p.speed * dt)
   278→   278→      if (p.owner === 0 || p.owner === 1) {
   279→   279→        // Edge crossings.
   280→   280→        let soundBudget = 2
   281→   281→        for (let i = 0; i < map.edges.length; i++) {
   282→   282→          if (p.hits[i]) continue
   283→   283→          const e = map.edges[i]
   284→   284→          const mx = (e.x1 + e.x2) / 2
   285→   285→          const my = (e.y1 + e.y2) / 2
   286→   286→          // Cheap reject by bounding box.
   287→   287→          const bx = Math.min(e.x1, e.x2) - p.r
   288→   288→          if (p.ox < bx || p.ox > Math.max(e.x1, e.x2) + p.r) {
   289→   289→            if (Math.abs(p.ox - mx) > p.r) continue
   290→   290→          }
   291→   291→          const d = segDist(p.ox, p.oy, e.x1, e.y1, e.x2, e.y2)
   292→   292→          if (d <= p.r && d > rPrev - 0.5) {
   293→   293→            p.hits[i] = 1
   294→   294→            const strength = clamp(1.08 - d / 720, 0.28, 1) * (p.owner === 1 ? 0.5 : 1)
   295→   295→            if (strength > this.revealS[i] * expFade(this.time - this.revealT[i], 0.55)) {
   296→   296→              this.revealS[i] = strength
   297→   297→              this.revealT[i] = this.time
   298→   298→            }
   299→   299→            if (p.owner === 0 && this.phase === 'playing' && soundBudget > 0 && this.thudGate <= 0) {
   300→   300→              const vol = strength * clamp(1.2 - d / 620, 0.2, 1) * 0.85
   301→   301→              this.audio.thud(this.panOf(mx), vol)
   302→   302→              soundBudget--
   303→   303→              this.thudGate = 0.045
   304→   304→            }
   305→   305→          }
   306→   306→        }
   307→   307→        // Door reveal.
   308→   308→        const dd = Math.hypot(map.door.midX - p.ox, map.door.midY - p.oy)
   309→   309→        if (dd <= p.r && dd > rPrev - 0.5) {
   310→   310→          this.doorRevealT = this.time
   311→   311→          if (p.owner === 0 && !p.doorHit && this.phase === 'playing') {
   312→   312→            p.doorHit = true
   313→   313→            this.audio.doorTone(this.panOf(map.door.midX), this.volByDist(dd, 640) * 0.5 + 0.05)
   314→   314→          }
   315→   315→        }
   316→   316→        // Key reveals.
   317→   317→        for (const k of this.keys) {
   318→   318→          if (k.found) continue
   319→   319→          const kd = Math.hypot(k.x - p.ox, k.y - p.oy)
   320→   320→          if (kd <= p.r && kd > rPrev - 0.5) {
   321→   321→            k.revealT = this.time
   322→   322→            if (p.owner === 0 && this.phase === 'playing') {
   323→   323→              this.audio.jingle(this.panOf(k.x), 0.16)
   324→   324→            }
   325→   325→          }
   326→   326→        }
   327→   327→        // Monster reveal (only player pings screech).
   328→   328→        const m = this.monster
   329→   329→        if (m && p.owner === 0) {
   330→   330→          const md = Math.hypot(m.x - p.ox, m.y - p.oy)
   331→   331→          if (md <= p.r && md > rPrev - 0.5) {
   332→   332→            m.revealT = this.time
   333→   333→            if (this.phase === 'playing' && this.screechGate <= 0) {
   334→   334→              this.audio.screech(this.panOf(m.x), clamp(0.6 - md / 1400, 0.22, 0.6))
   335→   335→              this.screechGate = 1.5
   336→   336→              if (m.state !== 'chase' && this.time > GRACE) {
   337→   337→                m.pendingAt = this.time + 0.3
   338→   338→                m.pendingX = this.player.x
   339→   339→                m.pendingY = this.player.y
   340→   340→              }
   341→   341→            }
   342→   342→          }
   343→   343→        }
   344→   344→      }
   345→   345→      if (p.r >= p.maxR) p.dead = true
   346→   346→    }
   347→   347→    this.pings = this.pings.filter((p) => !p.dead)
   348→   348→  }
   349→   349→
   350→   350→  private updatePlayer(dt: number): void {
   351→   351→    let { x, y } = this.input
   352→   352→    const l = Math.hypot(x, y)
   353→   353→    if (l > 1) {
   354→   354→      x /= l
   355→   355→      y /= l
   356→   356→    }
   357→   357→    const moving = l > 0.08
   358→   358→    const sp = PLAYER_SPEED
   359→   359→    const nx = this.player.x + x * sp * dt
   360→   360→    if (!this.hitsWall(nx, this.player.y)) this.player.x = nx
   361→   361→    const ny = this.player.y + y * sp * dt
   362→   362→    if (!this.hitsWall(this.player.x, ny)) this.player.y = ny
   363→   363→
   364→   364→    if (moving) {
   365→   365→      this.stepT -= dt
   366→   366→      if (this.stepT <= 0) {
   367→   367→        this.stepT = 0.34
   368→   368→        this.audio.step(0.045)
   369→   369→      }
   370→   370→    }
   371→   371→  }
   372→   372→
   373→   373→  private updateKeys(dt: number): void {
   374→   374→    for (const k of this.keys) {
   375→   375→      if (k.found) continue
   376→   376→      const d = Math.hypot(this.player.x - k.x, this.player.y - k.y)
   377→   377→      // Faint jingle hints when a key is close by.
   378→   378→      k.jT -= dt
   379→   379→      if (d < 300 && k.jT <= 0) {
   380→   380→        k.jT = 3.1 + Math.random() * 1.6
   381→   381→        this.audio.jingle(this.panOf(k.x), 0.05 + 0.1 * this.volByDist(d, 300))
   382→   382→      }
   383→   383→      if (d < 24) {
   384→   384→        k.found = true
   385→   385→        this.keysFound++
   386→   386→        this.audio.chime(this.panOf(k.x), 0.5)
   387→   387→        this.hooks.onKeys?.(this.keysFound)
   388→   388→        if (this.keysFound >= 3) {
   389→   389→          this.unlocked = true
   390→   390→          this.audio.unlock()
   391→   391→          this.hooks.onToast?.('THE DOOR IS OPEN')
   392→   392→          const m = this.monster
   393→   393→          if (m && m.state === 'dormant') m.state = 'wander'
   394→   394→        }
   395→   395→      }
   396→   396→    }
   397→   397→  }
   398→   398→
   399→   399→  private updateExit(dt: number): void {
   400→   400→    const map = this.map
   401→   401→    if (!map) return
   402→   402→    const d = Math.hypot(this.player.x - map.door.midX, this.player.y - map.door.midY)
   403→   403→    if (this.unlocked) {
   404→   404→      this.audio.setExitHum(this.volByDist(d, 280))
   405→   405→      this.beaconT -= dt
   406→   406→      if (this.beaconT <= 0) {
   407→   407→        this.beaconT = 2.4
   408→   408→        this.pings.push(this.makePing(map.door.midX, map.door.midY, 1, 170, 320))
   409→   409→        if (d < 620) this.audio.doorBlip(this.panOf(map.door.midX), 0.05 + 0.14 * this.volByDist(d, 620))
   410→   410→      }
   411→   411→      if (d < 42) {
   412→   412→        this.winAt = this.time
   413→   413→        this.phase = 'win'
   414→   414→        this.audio.setDanger(0)
   415→   415→        this.audio.setExitHum(0)
   416→   416→        this.audio.win()
   417→   417→        this.hooks.onEnd?.({
   418→   418→          outcome: 'escaped',
   419→   419→          timeMs: Math.round(this.time * 1000),
   420→   420→          keysFound: this.keysFound,
   421→   421→          pings: this.pingsUsed,
   422→   422→        })
   423→   423→      }
   424→   424→    } else {
   425→   425→      this.audio.setExitHum(0)
   426→   426→    }
   427→   427→  }
   428→   428→
   429→   429→  private updateDanger(dt: number): void {
   430→   430→    const m = this.monster
   431→   431→    if (!m) return
   432→   432→    if (this.time > GRACE) {
   433→   433→      if (!this.graceAnnounced) {
   434→   434→        this.graceAnnounced = true
   435→   435→        this.hooks.onToast?.('SOMETHING IS AWAKE')
   436→   436→        this.audio.growl(this.panOf(m.x), 0.5)
   437→   437→      }
   438→   438→      const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   439→   439→      let target = clamp(1 - d / 340, 0, 1)
   440→   440→      if (m.state !== 'chase') target *= 0.6
   441→   441→      this.danger += (target - this.danger) * Math.min(1, dt * 4)
   442→   442→      if (this.danger > 0.04) {
   443→   443→        this.hbT -= dt
   444→   444→        if (this.hbT <= 0) {
   445→   445→          this.hbT = 1.15 - 0.68 * this.danger
   446→   446→          this.audio.heartbeat(0.1 + 0.45 * this.danger)
   447→   447→        }
   448→   448→      }
   449→   449→      if (m.state === 'chase') this.shake = Math.max(this.shake, this.danger * 0.55)
   450→   450→    }
   451→   451→    this.audio.setDanger(this.danger)
   452→   452→  }
   453→   453→
   454→   454→  // -------------------------------------------------------------- monster
   455→   455→
   456→   456→  private updateMonster(dt: number): void {
   457→   457→    const map = this.map
   458→   458→    const m = this.monster
   459→   459→    if (!map || !m) return
   460→   460→    if (m.state === 'dormant') {
   461→   461→      if (this.time > GRACE) m.state = 'wander'
   462→   462→      return
   463→   463→    }
   464→   464→
   465→   465→    const d = Math.hypot(this.player.x - m.x, this.player.y - m.y)
   466→   466→    const mc = worldToCell(map, m.x, m.y)
   467→   467→
   468→   468→    // Sense the player up close — no ping needed.
   469→   469→    if (d < SENSE_R && m.state !== 'chase') {
   470→   470→      m.state = 'chase'
   471→   471→      m.path = []
   472→   472→      m.repath = 0
   473→   473→      m.lost = 0
   474→   474→      if (this.screechGate <= 0) {
   475→   475→        this.audio.screech(this.panOf(m.x), 0.65)
   476→   476→        this.screechGate = 1.2
   477→   477→      }
   478→   478→      if (!this.chaseAnnounced) {
   479→   479→        this.chaseAnnounced = true
   480→   480→        this.hooks.onToast?.('RUN')
   481→   481→      }
   482→   482→    }
   483→   483→
   484→   484→    // A ping it heard (delayed reaction).
   485→   485→    if (m.pendingAt > 0 && this.time >= m.pendingAt) {
   486→   486→      m.pendingAt = 0
   487→   487→      if (m.state !== 'chase') {
   488→   488→        const cell = nearestOpenCell(map, ...this.cellOfPoint(m.pendingX, m.pendingY))
   489→   489→        m.state = 'investigate'
   490→   490→        m.target = cell
   491→   491→        m.path = bfsPath(map, mc.c, mc.r, cell.c, cell.r)
   492→   492→        m.pathI = 0
   493→   493→      }
   494→   494→    }
   495→   495→
   496→   496→    let speed = 60
   497→   497→    if (m.state === 'chase') {
   498→   498→      speed = 148 + this.keysFound * 8
   499→   499→      m.lastPX = this.player.x
   500→   500→      m.lastPY = this.player.y
   501→   501→      if (d > LOSE_R) {
   502→   502→        m.lost += dt
   503→   503→        if (m.lost > 2.2) {
   504→   504→          m.state = 'investigate'
   505→   505→          m.target = nearestOpenCell(map, ...this.cellOfPoint(m.lastPX, m.lastPY))
   506→   506→          m.path = bfsPath(map, mc.c, mc.r, m.target.c, m.target.r)
   507→   507→          m.pathI = 0
   508→   508→        }
   509→   509→      } else {
   510→   510→        m.lost = 0
   511→   511→      }
   512→   512→      m.repath -= dt
   513→   513→      const pc = worldToCell(map, this.player.x, this.player.y)
   514→   514→      if (m.repath <= 0) {
   515→   515→        m.repath = 0.32
   516→   516→        m.path = bfsPath(map, mc.c, mc.r, pc.c, pc.r)
   517→   517→        m.pathI = 0
   518→   518→      }

   520→   520→      if (m.clickT <= 0) {
   521→   521→        m.clickT = 2.6
   522→   522→        this.monsterClick(m)
   523→   523→      }
   524→   524→      m.stepT -= dt
   525→   525→      if (m.stepT <= 0 && d < 460) {
   526→   526→        m.stepT = 0.27
   527→   527→        this.audio.monsterStep(this.panOf(m.x), 0.1 + 0.3 * this.volByDist(d, 460))
   528→   528→      }
   529→   529→    } else if (m.state === 'investigate') {
   530→   530→      speed = 98
   531→   531→      if (!m.target || m.pathI >= m.path.length) {
   532→   532→        m.state = 'wander'
   533→   533→        m.target = null
   534→   534→        m.path = []
   535→   535→      }
   536→   536→      m.clickT -= dt
   537→   537→      if (m.clickT <= 0) {
   538→   538→        m.clickT = 4.5 + Math.random() * 2.5
   539→   539→        this.monsterClick(m)
   540→   540→      }
   541→   541→    } else {
   542→   542→      // wander — it drifts through the dark, but its instinct pulls it
   543→   543→      // toward your region of the maze (biased random walk).
   544→   544→      speed = 60
   545→   545→      if (m.pathI >= m.path.length) {
   546→   546→        const cands: Cell[] = []
   547→   547→        for (let i = 0; i < 8; i++) {
   548→   548→          const c = clamp(mc.c + Math.floor((Math.random() - 0.5) * 22), 1, map.cols - 2)
   549→   549→          const r = clamp(mc.r + Math.floor((Math.random() - 0.5) * 16), 1, map.rows - 2)
   550→   550→          const cell = nearestOpenCell(map, c, r)
   551→   551→          if (Math.hypot(cell.c - mc.c, cell.r - mc.r) >= 3) cands.push(cell)
   552→   552→        }
   553→   553→        if (cands.length > 0) {
   554→   554→          let pick = cands[Math.floor(Math.random() * cands.length)]
   555→   555→          if (Math.random() < 0.6) {
   556→   556→            const pc = worldToCell(map, this.player.x, this.player.y)
   557→   557→            for (const c of cands) {
   558→   558→              if (
   559→   559→                Math.hypot(c.c - pc.c, c.r - pc.r) <
   560→   560→                Math.hypot(pick.c - pc.c, pick.r - pc.r)
   561→   561→              ) {
   562→   562→                pick = c
   563→   563→              }
   564→   564→            }
   565→   565→          }
   566→   566→          m.path = bfsPath(map, mc.c, mc.r, pick.c, pick.r)
   567→   567→          m.pathI = 0
   568→   568→        }
   569→   569→      }
   570→   570→      m.growlT -= dt
   571→   571→      if (m.growlT <= 0) {
   572→   572→        m.growlT = 3.5 + Math.random() * 4.5
   573→   573→        this.audio.growl(this.panOf(m.x), 0.06 + 0.4 * this.volByDist(d, 560))
   574→   574→      }
   575→   575→      m.clickT -= dt
   576→   576→      if (m.clickT <= 0) {
   577→   577→        m.clickT = 6 + Math.random() * 3
   578→   578→        this.monsterClick(m)
   579→   579→      }
   580→   580→    }
   581→   581→
   582→   582→    // Follow path.
   583→   583→    if (m.pathI < m.path.length) {
   584→   584→      const wp = m.path[m.pathI]
   585→   585→      const dx = wp.x - m.x
   586→   586→      const dy = wp.y - m.y
   587→   587→      const dl = Math.hypot(dx, dy)
   588→   588→      if (dl < 6) {
   589→   589→        m.pathI++
   590→   590→      } else {
   591→   591→        m.x += (dx / dl) * speed * dt
   592→   592→        m.y += (dy / dl) * speed * dt
   593→   593→      }
   594→   594→    }
   595→   595→
   596→   596→    // Catch.
   597→   597→    if (d < CATCH_R) {
   598→   598→      this.deathAt = this.time
   599→   599→      this.phase = 'dead'
   600→   600→      this.shake = 1.4
   601→   601→      this.audio.setDanger(0)
   602→   602→      this.audio.setExitHum(0)
   603→   603→      this.audio.death()
   604→   604→      this.hooks.onEnd?.({
   605→   605→        outcome: 'caught',
   606→   606→        timeMs: Math.round(this.time * 1000),
   607→   607→        keysFound: this.keysFound,
   608→   608→        pings: this.pingsUsed,
   609→   609→      })
   610→   610→    }
   611→   611→  }
